1//! A buffer which keeps only the end of what got written to it.
2
3/// Keeps the last `MAX` bytes of what got pushed, like the end of the stderr
4/// of a runtime, which holds its error.
5#[derive(Debug, Default)]
6pub struct Tail<const MAX: usize> {
7    buf: Vec<u8>,
8}
9
10impl<const MAX: usize> Tail<MAX> {
11    pub fn push(&mut self, data: &[u8]) {
12        self.buf.extend_from_slice(data);
13        // Dropping the front only once it doubled keeps it amortized.
14        if self.buf.len() > 2 * MAX {
15            self.buf.drain(..self.buf.len() - MAX);
16        }
17    }
18
19    /// The last `MAX` bytes.
20    pub fn end(&self) -> &[u8] {
21        &self.buf[self.buf.len().saturating_sub(MAX)..]
22    }
23
24    /// The last `MAX` bytes, without copying them.
25    pub fn into_end(mut self) -> Vec<u8> {
26        let excess = self.buf.len().saturating_sub(MAX);
27        self.buf.drain(..excess);
28        self.buf
29    }
30}
31
32#[cfg(test)]
33mod tests {
34    use super::*;
35
36    #[test]
37    fn keeps_the_end() {
38        let mut tail = Tail::<4>::default();
39        tail.push(b"ab");
40        assert_eq!(tail.end(), b"ab");
41        for i in 0..100u8 {
42            tail.push(&[i]);
43            assert!(tail.buf.len() <= 8);
44        }
45        assert_eq!(tail.end(), [96, 97, 98, 99]);
46        assert_eq!(tail.into_end(), [96, 97, 98, 99]);
47    }
48}
