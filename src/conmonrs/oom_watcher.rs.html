48/// The OOM related counters of a memory events file.
49#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
50struct OomCounters {
51    /// Times the memory limit of the cgroup was hit.
52    oom: u64,
53
54    /// Processes killed because of it. Kills are accounted to the victim's
55    /// leaf cgroup, so this drifts apart from `oom` by design and the two are
56    /// compared separately.
57    oom_kill: u64,
58}
59
60impl OomCounters {
61    /// Whether any counter increased compared to `baseline`.
62    fn increased_since(&self, baseline: &Self) -> bool {
63        self.oom > baseline.oom || self.oom_kill > baseline.oom_kill
64    }
65}

67#[derive(Debug, Default, PartialEq, Eq)]
68pub struct OOMEvent {
69    pub oom: bool,
70
71    /// The cgroup of the process got removed, which hides whether an OOM
72    /// happened, but there is a hint which may also stem from a sibling:
73    /// with cgroup v1, an event before the removal, with cgroup v2, a new
74    /// OOM kill of an outer cgroup or none to tell. So it only counts for a
75    /// process which got killed by SIGKILL, see [`OOMEvent::is_oom`].
76    pub possible_oom: bool,
77}

79impl OOMEvent {
80    fn new(oom: bool) -> Self {
81        Self {
82            oom,
83            ..Self::default()
84        }
85    }
86
87    /// A possible OOM if `possible`, none otherwise.
88    pub(crate) fn possible(possible: bool) -> Self {

90            oom: false,
91            possible_oom: possible,

101/// The state of a cgroup v1 `memory.oom_control` file.
102#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
103struct OomControl {
104    /// Whether the OOM killer is disabled, tasks wait in the OOM then.
105    kill_disable: bool,
106
107    /// Whether the cgroup is in an OOM right now.
108    under_oom: bool,
109
110    /// Processes killed in this cgroup, kernels before 4.13 have no counter.
111    oom_kill: Option<u64>,
112}
113
114impl OomControl {
115    fn parse(content: &str) -> Self {
116        let mut control = Self::default();
117        for line in content.lines() {
118            let Some((key, value)) = line.split_once(' ') else {
119                continue;
120            };
121            let value = value.trim().parse::<u64>().ok();
122            match key {
123                "oom_kill_disable" => control.kill_disable = value.is_some_and(|v| v != 0),
124                "under_oom" => control.under_oom = value.is_some_and(|v| v != 0),
125                "oom_kill" => control.oom_kill = value,
126                _ => {}
127            }
128        }
129        control
130    }
131}

180    async fn oom_handling_cgroup_v1(
181        token: CancellationToken,
182        pid: u32,
183        exit_paths: &[PathBuf],
184        tx: Sender<OOMEvent>,
185    ) -> Result<()> {
186        // The setup reads procfs and cgroupfs inline, which does not block,
187        // instead of adding a blocking task to every container and exec.
188        let memory_cgroup_path = if let Some(path) =
189            Self::process_cgroup_subsystem_path_cgroup_v1(pid, "memory")
190                .context("process cgroup memory subsystem path")?
191        {
192            path
193        } else {
194            debug!("Stopping OOM handler because no cgroup subsystem path exists");
195            return Ok(());
196        };
197
198        debug!(
199            "Using memory cgroup v1 path: {}",
200            memory_cgroup_path.display()
201        );
202
203        let memory_cgroup_file_oom_path = memory_cgroup_path.join("memory.oom_control");
204        let event_control_path = memory_cgroup_path.join("cgroup.event_control");
205        let path = memory_cgroup_file_oom_path.to_str();
206
207        debug!(path, "Setup cgroup v1 oom handling");
208
209        // Taken before registering, an OOM in between is found by the check on
210        // exit. Unknown without the counter or if it cannot be read, then
211        // every event counts.
212        let baseline = match Self::read_oom_control(&memory_cgroup_file_oom_path) {
213            Ok(control) => control.oom_kill,
214            Err(e) => {
215                debug!("Could not read initial OOM kills: {:#}", e);
216                None
217            }
218        };
219
220        // Registering an event only needs the file open, like for conmon.
221        let oom_cgroup_file = File::open(&memory_cgroup_file_oom_path)
222            .await
223            .context("opening cgroup oom file")?;
224        let oom_event_fd = AsyncEventFd::new().context("creating eventfd")?;
225
226        let mut event_control = OpenOptions::new()
227            .write(true)
228            .open(event_control_path)
229            .await
230            .context("opening cgroup event control file")?;
231        event_control
232            .write_all(
233                format!(
234                    "{} {}",
235                    oom_event_fd.as_raw_fd(),
236                    oom_cgroup_file.as_raw_fd()
237                )
238                .as_bytes(),
239            )
240            .await
241            .context("writing control data")?;
242        event_control.flush().await.context("flush control data")?;
243
244        debug!("Successfully setup cgroup v1 oom detection");
245
246        let event =
247            Self::watch_cgroup_v1(&token, oom_event_fd, &memory_cgroup_file_oom_path, baseline)
248                .await;
249        if event.oom {
250            Self::write_oom_files(exit_paths).await;
251        }
252        if let Err(e) = tx.try_send(event) {
253            error!("try_send failed: {:#}", e);
254        }
255
256        debug!("Done watching for ooms");
257        Ok(())
258    }

260    /// Wait for an OOM of a cgroup v1 memory controller until `token` gets
261    /// cancelled and return what was found.
262    ///
263    /// The eventfd also fires for an OOM of an ancestor cgroup, whose victim
264    /// may be a sibling container, and once when the cgroup gets removed. So
265    /// an event gets confirmed by the `oom_kill` counter of the cgroup. The
266    /// kernel signals the eventfd before it kills, so the counter is read
267    /// again a few times. It is not hierarchical either, a victim in a memcg
268    /// nested inside the container is not counted, which `under_oom` covers.
269    /// A missed OOM is worse than a rare false one, like conmon: an event
270    /// counts without a known baseline or counter, and if the process exits
271    /// while the event gets confirmed and the cgroup is still under OOM. An
272    /// event which stayed unconfirmed, like an OOM of a sibling, does not
273    /// count later, a normal stop would be taken for an OOM otherwise. If the
274    /// cgroup goes away after an event, it may have been an OOM which killed
275    /// the last process or the one of a sibling followed by a normal exit,
276    /// see [`OOMEvent::possible_oom`]. The removal alone signals once.
277    async fn watch_cgroup_v1(
278        token: &CancellationToken,
279        mut event_fd: impl AsyncRead + Unpin,
280        oom_control: &Path,
281        baseline: Option<u64>,
282    ) -> OOMEvent {
283        let mut buffer = [0u8; 8];
284        loop {
285            tokio::select! {
286                // An OOM kill often ends the process, which cancels the token.
287                // Handle the event first then.
288                biased;
289                res = event_fd.read(&mut buffer) => {
290                    let count = match res {
291                        Ok(n) if n == buffer.len() => u64::from_ne_bytes(buffer),
292                        res => {
293                            debug!("Unable to read OOM eventfd, checking on exit only: {res:?}");
294                            break;
295                        }
296                    };
297                    debug!("Got OOM event, count {count}");
298                    match Self::confirm_oom_event(token, oom_control, baseline).await {
299                        Confirmation::Oom => return OOMEvent::new(true),
300                        Confirmation::Removed => {
301                            // Removing the cgroup signals the eventfd once. A
302                            // count above that means another event came
303                            // before.
304                            debug!("Memory cgroup removed");
305                            return OOMEvent::possible(count > 1);
306                        }
307                        Confirmation::Gone => {
308                            debug!("Memory cgroup removed while confirming the event");
309                            return OOMEvent::possible(true);
310                        }
311                        Confirmation::Unconfirmed => {
312                            debug!("OOM event without a kill in this cgroup");
313                        }
314                        Confirmation::Exited => {
315                            return Self::check_oom_on_exit(oom_control, baseline, true).await;
316                        }
317                    }
318                }
319                _ = token.cancelled() => {
320                    debug!("Loop cancelled");
321                    break;
322                }
323            }
324        }
325        token.cancelled().await;
326        Self::check_oom_on_exit(oom_control, baseline, false).await
327    }

337    /// Confirm an eventfd event of a cgroup v1 memory controller, see
338    /// [`OOMWatcher::watch_cgroup_v1`].
339    async fn confirm_oom_event(
340        token: &CancellationToken,
341        oom_control: &Path,
342        baseline: Option<u64>,
343    ) -> Confirmation {
344        let Some(baseline) = baseline else {
345            debug!("No OOM kill baseline, taking the event as OOM");
346            return Confirmation::Oom;
347        };
348        let mut delays = Self::CONFIRM_BACKOFF.into_iter();
349        let mut first = true;
350        loop {
351            match Self::read_oom_control(oom_control) {
352                Err(e) if e.kind() == ErrorKind::NotFound => {
353                    return if first {
354                        Confirmation::Removed
355                    } else {
356                        Confirmation::Gone
357                    };
358                }
359                Err(e) => {
360                    debug!("Unable to read OOM kills, taking the event as OOM: {:#}", e);
361                    return Confirmation::Oom;
362                }
363                Ok(control) => match control.oom_kill {
364                    Some(kills) if kills > baseline => return Confirmation::Oom,
365                    None => return Confirmation::Oom,
366                    // Without the OOM killer the tasks wait in the OOM.
367                    Some(_) if control.kill_disable && control.under_oom => {
368                        return Confirmation::Oom;
369                    }
370                    Some(_) => match delays.next() {
371                        Some(delay) => {
372                            first = false;
373                            tokio::select! {
374                                _ = time::sleep(delay) => {}
375                                // The counter is final once the process exited.
376                                _ = token.cancelled() => return Confirmation::Exited,
377                            }
378                        }
379                        // A victim in a nested memcg, which the counter of
380                        // this cgroup does not count.
381                        None if control.under_oom => return Confirmation::Oom,
382                        None => return Confirmation::Unconfirmed,
383                    },
384                },
385            }
386        }
387    }

389    /// Check for an OOM kill a last time once the process exited. An event
390    /// whose confirmation the exit interrupted counts if the cgroup is still
391    /// under OOM, and may have been one if the cgroup is gone.
392    async fn check_oom_on_exit(
393        oom_control: &Path,
394        baseline: Option<u64>,
395        confirming: bool,
396    ) -> OOMEvent {
397        match Self::read_oom_control(oom_control) {
398            // Without a baseline the counter tells nothing, it may stem from
399            // before the container. An event took that into account already.
400            Ok(control) => OOMEvent::new(
401                baseline
402                    .zip(control.oom_kill)
403                    .is_some_and(|(baseline, kills)| kills > baseline)
404                    || (confirming && control.under_oom),
405            ),
406            Err(e) if e.kind() == ErrorKind::NotFound => OOMEvent::possible(confirming),
407            Err(e) => {
408                debug!("Unable to read OOM kills on exit: {:#}", e);
409                OOMEvent::new(confirming)
410            }
411        }
412    }

455    /// Wait for an OOM of the cgroup v2 `leaf` until `token` gets cancelled
456    /// and return what was found.
457    ///
458    /// An OOM counts once the counters of the leaf, or of the cgroup which
459    /// merely wraps it, went up. The leaf may be gone at the exit, the
460    /// systemd cgroup manager removes the scope of a container once its
461    /// processes exited. Then the hierarchical `oom_kill` counter of the
462    /// nearest ancestor which outlives it tells whether a kill happened
463    /// meanwhile, see [`OOMEvent::possible_oom`]. A removal is no OOM
464    /// on its own. cgroupfs reports it with `IN_IGNORED` only, which the
465    /// watcher drops, the remove event shows up for other file systems.
466    async fn watch_cgroup_v2(token: &CancellationToken, leaf: &Path) -> OOMEvent {
467        let mut cgroup = CgroupV2::new(leaf);
468
469        let path = cgroup.events.path.to_str();
470        debug!(path, "Setup cgroup v2 handling");
471
472        let mut subscriptions = match Self::subscribe(
473            &cgroup.events,
474            cgroup.parent.as_ref(),
475            cgroup.outer.as_ref().map(|(path, _)| path.as_path()),
476        ) {
477            Ok(subscriptions) => subscriptions,
478            // The cgroup may be gone already. Still check for an OOM once the
479            // process exited.
480            Err(e) => {
481                debug!(
482                    "Unable to watch memory events, checking on exit only: {:#}",
483                    e
484                );
485                token.cancelled().await;
486                return cgroup.check_on_exit();
487            }
488        };
489
490        let mut restarts = 0;
491        loop {
492            tokio::select! {
493                // Memory events may change all the time, for example while
494                // processes left in the cgroup hit memory.high, which must
495                // not hold back the exit. The check on exit reads the
496                // counters of an OOM which ended the process anyway.
497                biased;
498                _ = token.cancelled() => {
499                    debug!("Loop cancelled");
500                    return cgroup.check_on_exit();
501                }
502                event = subscriptions.rx.recv() => {
503                    let Some(event) = event else {
504                        // All watches ended, their files got removed, or the
505                        // watcher stopped reading events.
506                        let path = cgroup.events.path.to_str();
507                        if !subscriptions.watcher_failed() || restarts >= Self::MAX_WATCHER_RESTARTS {
508                            debug!(path, "Watches ended, checking once the process exited");
509                            break;
510                        }
511                        restarts += 1;
512                        warn!(path, "Watcher of the memory events stopped, watching again");
513                        match Self::subscribe(
514                            &cgroup.events,
515                            cgroup.parent.as_ref(),
516                            cgroup.outer.as_ref().map(|(path, _)| path.as_path()),
517                        ) {
518                            Ok(new) => subscriptions = new,
519                            Err(e) => {
520                                warn!(path, "Unable to watch memory events again, checking on exit only: {:#}", e);
521                                break;
522                            }
523                        }
524                        // The changes while nothing got watched, like the
525                        // kills of siblings, refresh the baselines.
526                        if cgroup.check(true) {
527                            return OOMEvent::new(true);
528                        }
529                        continue;
530                    };
531                    debug!("Got OOM file event: {:?}", event);
532                    if event.removed {
533                        debug!("Got remove event, checking once the process exited");
534                        break;
535                    }
536
537                    if cgroup.check(cgroup.is_outer_event(&event)) {
538                        return OOMEvent::new(true);
539                    }
540                    if Self::file_not_found(&cgroup.events.path) {
541                        debug!("Memory events file gone, checking once the process exited");
542                        break;
543                    }
544                    debug!("No OOM found in event");
545                }
546            }
547        }
548
549        drop(subscriptions);
550        token.cancelled().await;
551        cgroup.check_on_exit()
552    }

558    /// Watch the memory events of the container and, if present, the ones of
559    /// its wrapper cgroup.
560    fn subscribe(
561        events: &MemoryEvents,
562        parent_events: Option<&MemoryEvents>,
563        pod_events: Option<&Path>,
564    ) -> Result<Subscriptions> {
565        let watcher = SharedWatcher::get()?;
566        let (events_tx, rx) = channel(WATCH_EVENT_CHANNEL_SIZE);
567        let subscription = watcher
568            .subscribe(&events.path, events_tx.clone())
569            .context("watch memory events file")?;
570
571        let mut parent_subscription = None;
572        if let Some(parent) = parent_events {
573            debug!(
574                "Also watching parent memory events: {}",
575                parent.path.display()
576            );
577            match watcher.subscribe(&parent.path, events_tx.clone()) {
578                Ok(subscription) => parent_subscription = Some(subscription),
579                Err(e) => debug!("Could not watch parent memory events: {:#}", e),
580            }
581        }
582
583        // The kills of siblings advance the baseline of the outer cgroup.
584        let mut pod_subscription = None;
585        if let Some(pod) = pod_events {
586            debug!("Also watching outer memory events: {}", pod.display());
587            match watcher.subscribe(pod, events_tx) {
588                Ok(subscription) => pod_subscription = Some(subscription),
589                Err(e) => debug!("Could not watch pod memory events: {:#}", e),
590            }
591        }
592
593        Ok(Subscriptions {
594            rx,
595            subscription,
596            _parent_subscription: parent_subscription,
597            _pod_subscription: pod_subscription,
598        })
599    }

601    /// Check the memory events of the container and of its wrapper cgroup for
602    /// new OOMs. An OOM of the parent only counts while it wraps the container
603    /// alone, otherwise it may stem from a sibling.
604    fn check_cgroups(
605        leaf: &Path,
606        events: &mut MemoryEvents,
607        parent_events: Option<&mut MemoryEvents>,
608    ) -> bool {
609        if events.check() {
610            return true;
611        }
612        let Some(parent) = parent_events else {
613            return false;
614        };
615        // Only list the parent once its counters went up, memory events change
616        // often under memory pressure. The counters are the baseline for the
617        // next check either way, so an OOM skipped here is not reported later.
618        if !parent.check() {
619            return false;
620        }
621        if Self::is_only_child(leaf) {
622            return true;
623        }
624        debug!("Ignoring OOM of parent cgroup with siblings");
625        false
626    }

661    /// Whether the parent of `leaf` contains no other cgroup than `leaf`, so
662    /// that it merely wraps it. The memory limit cannot tell the layouts
663    /// apart, crun applies it to both the wrapper and the leaf. A parent
664    /// without any child still counts, the leaf may already be gone when
665    /// checking for an OOM after the exit.

684    fn check_for_oom(

686        last: OomCounters,

693        let found_oom = counters.increased_since(&last);

698    /// Parse the `oom` and `oom_kill` counters of a memory events file.
699    fn read_oom_counters(path: &Path) -> Result<OomCounters> {
700        let fp = StdFile::open(path)
701            .with_context(|| format!("open memory events file: {}", path.display()))?;
702        let mut counters = OomCounters::default();
703        let mut has_oom = false;
704
705        for line in BufReader::new(fp).lines() {
706            let line = line.context("read line from memory events file")?;
707            trace!(line);
708
709            let Some((key, value)) = line.split_once(' ') else {
710                continue;
711            };
712            let target = match key {
713                "oom" => {
714                    has_oom = true;
715                    &mut counters.oom
716                }
717                "oom_kill" => &mut counters.oom_kill,
718                _ => continue,
719            };
720            *target = value.trim_end().parse().context("parse u64 counter")?;
721        }
722
723        // Every memory events file has it, also of kernels without the kill
724        // counter. Without, it is no memory events file, or one which is
725        // being written, like a fake one.
726        if !has_oom {
727            bail!("no oom counter in memory events file {}", path.display())
728        }
729        Ok(counters)
730    }

798    /// `hierarchy-ID:controller-list:cgroup-path` is the one of its hierarchy.
799    /// The path may contain colons itself. The unified hierarchy of a hybrid
800    /// setup lists no controllers, it is not the one of any subsystem.
801    fn cgroup_v1_path(line: &str, subsystem: &str) -> Result<Option<PathBuf>> {

807            return Ok(Some(Path::new(CGROUP_ROOT).join(subsystem).join(path)));
808        }
809        Ok(None)

844/// The memory events of a cgroup v2 leaf and of the cgroups above it.
845struct CgroupV2 {
846    leaf: PathBuf,
847    events: MemoryEvents,
848
849    /// The cgroup which merely wraps the leaf, crun places the process in
850    /// a leaf below the container cgroup and the OOMs end up recorded in
851    /// that parent. Only follow the parent in that layout. A pod cgroup
852    /// shared with sibling containers (runc) must not be watched, otherwise
853    /// an OOM of a sibling would be reported for this container. Prefer
854    /// memory.events.local, the counters in memory.events include events
855    /// from below, and fall back on kernels without the local file.
856    parent: Option<MemoryEvents>,
857
858    /// The hierarchical memory events of the pod cgroup, which outlives a
859    /// leaf the systemd cgroup manager removes, or of the parent without
860    /// one, and the OOM counters which the leaf does not account for. They
861    /// advance while the leaf exists, so that only kills after its removal
862    /// stay ambiguous.
863    outer: Option<(PathBuf, OomCounters)>,
864}
865
866impl CgroupV2 {
867    /// How many levels above the leaf the pod cgroup may be: its parent with
868    /// runc, its grandparent with crun, which adds a leaf below the
869    /// container cgroup.
870    const POD_LEVELS: usize = 2;
871
872    fn new(leaf: &Path) -> Self {
873        let outer = Self::pod_cgroup(leaf)
874            .or_else(|| leaf.parent())
875            .and_then(|outer| {
876                let path = outer.join("memory.events");
877                match OOMWatcher::read_oom_counters(&path) {
878                    Ok(counters) => Some((path, counters)),
879                    Err(e) => {
880                        debug!("No OOM baseline of the outer cgroup: {:#}", e);
881                        None
882                    }
883                }
884            });
885        Self {
886            leaf: leaf.to_path_buf(),
887            events: MemoryEvents::new(leaf.join("memory.events")),
888            parent: OOMWatcher::wrapper_events_path(leaf).map(MemoryEvents::new),
889            outer,
890        }
891    }
892
893    /// Whether a watch event concerns the outer cgroup. Events which the
894    /// kernel dropped may have, the baseline gets read again then.
895    fn is_outer_event(&self, event: &WatchEvent) -> bool {
896        self.outer.as_ref().is_some_and(|(path, _)| {
897            event
898                .path
899                .as_deref()
900                .is_none_or(|event_path| event_path == path)
901        })
902    }
903
904    /// The pod cgroup above the leaf, if there is one within
905    /// [`CgroupV2::POD_LEVELS`]. Further up, like in a QoS cgroup, the
906    /// siblings are other pods.
907    fn pod_cgroup(leaf: &Path) -> Option<&Path> {
908        leaf.ancestors()
909            .skip(1)
910            .take(Self::POD_LEVELS)
911            .find(|ancestor| {
912                ancestor
913                    .file_name()
914                    .and_then(|name| name.to_str())
915                    .is_some_and(Self::is_pod_cgroup_name)
916            })
917    }
918
919    /// Whether the name is the one the kubelet gives to a pod cgroup, like
920    /// `pod<uid>` with the cgroupfs driver and
921    /// `kubepods-burstable-pod<uid>.slice` with the systemd one.
922    fn is_pod_cgroup_name(name: &str) -> bool {
923        let is_pod = |name: &str| {
924            name.strip_prefix("pod").is_some_and(|uid| {
925                !uid.is_empty()
926                    && uid
927                        .chars()
928                        .all(|c| c.is_ascii_hexdigit() || c == '-' || c == '_')
929            })
930        };
931        match name.strip_suffix(".slice") {
932            Some(slice) => slice.rsplit('-').next().is_some_and(is_pod),
933            None => is_pod(name),
934        }
935    }
936
937    /// Whether the counters of the leaf or its wrapper went up. After an
938    /// event of the outer cgroup, its kills since the last check stem from
939    /// a sibling otherwise, as long as the leaf still exists. The kernel
940    /// counts a kill in the leaf before its ancestors, so the outer cgroup
941    /// gets read first.
942    fn check(&mut self, outer_changed: bool) -> bool {
943        let outer = self
944            .outer
945            .as_ref()
946            .filter(|_| outer_changed)
947            .and_then(|(path, _)| OOMWatcher::read_oom_counters(path).ok());
948        if OOMWatcher::check_cgroups(&self.leaf, &mut self.events, self.parent.as_mut()) {
949            return true;
950        }
951        if let (Some(counters), Some((_, baseline))) = (outer, self.outer.as_mut())
952            && !OOMWatcher::file_not_found(&self.events.path)
953        {
954            *baseline = counters;
955        }
956        false
957    }
958
959    /// Check for an OOM a last time once the process exited. If the leaf is
960    /// gone, a kill of the outer cgroup since tells that it may have been
961    /// one, and so does an outer cgroup which is gone as well.
962    fn check_on_exit(&mut self) -> OOMEvent {
963        debug!("Last resort check for OOM");
964        if self.check(false) {
965            return OOMEvent::new(true);
966        }
967        if !OOMWatcher::file_not_found(&self.events.path) {
968            return OOMEvent::default();
969        }
970        let Some((path, baseline)) = &self.outer else {
971            debug!("Leaf cgroup gone without outer counters");
972            return OOMEvent::possible(true);
973        };
974        match OOMWatcher::read_oom_counters(path) {
975            Ok(counters) => {
976                debug!(
977                    "Leaf cgroup gone, OOM kills of {}: {} since {}",
978                    path.display(),
979                    counters.oom_kill,
980                    baseline.oom_kill
981                );
982                OOMEvent::possible(counters.oom_kill > baseline.oom_kill)
983            }
984            Err(e) => {
985                debug!("Unable to read outer OOM counters: {:#}", e);
986                OOMEvent::possible(true)
987            }
988        }
989    }
990}

1042/// A watch event as delivered to subscribers.
1043#[derive(Clone, Debug, PartialEq, Eq)]
1044struct WatchEvent {
1045    /// The file which changed. None if the kernel dropped events, every
1046    /// watched file may have changed then.
1047    path: Option<Arc<Path>>,
1048
1049    /// Whether the file got removed, which ends its watch.
1050    removed: bool,
1051}
1052
1053/// What a watch reports. Opening a file is no change, the watcher reads the
1054/// files itself.
1055const WATCH_FLAGS: AddWatchFlags = AddWatchFlags::IN_MODIFY
1056    .union(AddWatchFlags::IN_ATTRIB)
1057    .union(AddWatchFlags::IN_CLOSE_WRITE)
1058    .union(AddWatchFlags::IN_DELETE_SELF)
1059    .union(AddWatchFlags::IN_MOVE_SELF);
1060
1061/// The watches by descriptor. A watch belongs to a file rather than to its
1062/// path: a file which got recreated gets a watch of its own, and the events
1063/// of the old one, like its removal, must not reach the subscribers of the
1064/// new one.
1065#[derive(Default)]
1066struct Watches(HashMap<WatchDescriptor, Watch>);
1067
1068/// A watch of a file and its subscribers.
1069struct Watch {
1070    path: Arc<Path>,
1071    subscribers: Vec<(u64, Sender<WatchEvent>)>,
1072}
1073
1074impl Watches {
1075    /// Hand an event of the kernel to the subscribers of its watch.
1076    fn dispatch(&mut self, event: &InotifyEvent) {
1077        if event.mask.contains(AddWatchFlags::IN_Q_OVERFLOW) {
1078            self.lost();
1079            return;
1080        }
1081        let Some(watch) = self.0.get(&event.wd) else {
1082            trace!("Watch event with unknown descriptor: {event:?}");
1083            return;
1084        };
1085        // A removal of a cgroup only removes the watch, which tells nothing
1086        // about the file, see [`OOMWatcher::watch_cgroup_v2`].
1087        if event.mask.intersects(WATCH_FLAGS) {
1088            let watch_event = WatchEvent {
1089                path: Some(watch.path.clone()),
1090                removed: event.mask.contains(AddWatchFlags::IN_DELETE_SELF),
1091            };
1092            for (_, tx) in &watch.subscribers {
1093                Self::deliver(tx, &watch_event);
1094            }
1095        }
1096        // The last event of a watch, the kernel may reuse the descriptor.
1097        if event.mask.contains(AddWatchFlags::IN_IGNORED) {
1098            self.0.remove(&event.wd);
1099        }
1100    }
1101
1102    /// Tell every subscriber that events got lost, which may concern every
1103    /// file.
1104    fn lost(&self) {
1105        let lost = WatchEvent {
1106            path: None,
1107            removed: false,
1108        };
1109        for (_, tx) in self.0.values().flat_map(|watch| &watch.subscribers) {
1110            Self::deliver(tx, &lost);
1111        }
1112    }
1113
1114    /// Deliver an event without ever blocking the reader of the events. A
1115    /// full channel drops the event, the receiver re-reads the counters on
1116    /// the next one anyway.
1117    fn deliver(tx: &Sender<WatchEvent>, event: &WatchEvent) {
1118        if tx.try_send(event.clone()).is_err() {
1119            trace!("Dropping watch event, subscriber not ready");
1120        }
1121    }
1122}

1133/// A single inotify instance shared by all OOM watchers.
1134///
1135/// Creating an inotify instance per container does not scale with the number
1136/// of containers in a pod. All memory event files are watched through this
1137/// instance instead, a task reads its events and dispatches them to the
1138/// subscribers of their path. Adding and removing a watch does not block, the
1139/// subscribers do that right away.
1140struct SharedWatcher {
1141    inotify: Arc<AsyncFd<InotifyFd>>,
1142
1143    /// The subscribers and the watches. The reader of the events takes this
1144    /// lock as well, it never waits while holding it.
1145    watches: Arc<Mutex<Watches>>,
1146
1147    /// Set once the reader of the events stopped, the next subscription gets
1148    /// a new watcher then.
1149    failed: Arc<AtomicBool>,
1150
1151    /// The task which reads the events, it ends with the watcher.
1152    reader: JoinHandle<()>,
1153
1154    /// Source of unique subscription identifiers.
1155    next_id: AtomicU64,
1156}
1157
1158/// The shared watcher is kept alive by its subscriptions. It gets created on
1159/// the first subscription and dropped with the last one, which frees the
1160/// inotify instance while no container is watched. A failed creation is
1161/// retried on the next subscription instead of being cached, and so is a
1162/// watcher whose reader stopped. The server has a single runtime, which reads
1163/// the events.
1164#[cfg(not(test))]
1165static SHARED_WATCHER: Mutex<Weak<SharedWatcher>> = Mutex::new(Weak::new());
1166
1167/// The unit tests run a runtime per test, each one gets a watcher of its own,
1168/// which its runtime reads.
1169#[cfg(test)]
1170static SHARED_WATCHERS: Mutex<Vec<(tokio::runtime::Id, Weak<SharedWatcher>)>> =
1171    Mutex::new(Vec::new());
1172
1173impl SharedWatcher {
1174    /// Retrieve the shared watcher, creating it if necessary.
1175    fn get() -> Result<Arc<Self>> {
1176        // The reader of the events is a task of the runtime.
1177        let runtime = Handle::try_current().context("get runtime")?;
1178        Self::with_slot(&runtime, |slot| {
1179            if let Some(watcher) = slot
1180                .upgrade()
1181                .filter(|watcher| !watcher.failed.load(Ordering::Acquire))
1182            {
1183                return Ok(watcher);
1184            }
1185            let watcher = Arc::new(Self::new().context("create shared file watcher")?);
1186            *slot = Arc::downgrade(&watcher);
1187            Ok(watcher)
1188        })
1189    }
1190
1191    #[cfg(not(test))]
1192    fn with_slot<T>(_: &Handle, f: impl FnOnce(&mut Weak<Self>) -> Result<T>) -> Result<T> {
1193        f(&mut *lock(&SHARED_WATCHER)?)
1194    }
1195
1196    #[cfg(test)]
1197    fn with_slot<T>(runtime: &Handle, f: impl FnOnce(&mut Weak<Self>) -> Result<T>) -> Result<T> {
1198        let id = runtime.id();
1199        let mut shared = lock(&SHARED_WATCHERS)?;
1200        shared.retain(|(_, watcher)| watcher.strong_count() > 0);
1201        let pos = match shared.iter().position(|(i, _)| *i == id) {
1202            Some(pos) => pos,
1203            None => {
1204                shared.push((id, Weak::new()));
1205                shared.len() - 1
1206            }
1207        };
1208        f(&mut shared[pos].1)
1209    }
1210
1211    #[allow(unsafe_code)]
1212    fn new() -> Result<Self> {
1213        let inotify = Inotify::init(InitFlags::IN_NONBLOCK | InitFlags::IN_CLOEXEC)
1214            .context("create inotify instance")?;
1215        // SAFETY: the `AsyncFd` takes ownership of the inotify instance, so it
1216        // stays valid and unchanged for the lifetime of the `AsyncFd`.
1217        let inotify =
1218            unsafe { AsyncFd::register_with_interest(InotifyFd(inotify), Interest::READABLE) }
1219                .context("register inotify instance")?;
1220        let inotify = Arc::new(inotify);
1221        let watches = Arc::<Mutex<Watches>>::default();
1222        let failed = Arc::<AtomicBool>::default();
1223        let reader = task::spawn(Self::read_events(
1224            inotify.clone(),
1225            watches.clone(),
1226            failed.clone(),
1227        ));
1228        Ok(Self {
1229            inotify,
1230            watches,
1231            failed,
1232            reader,
1233            next_id: AtomicU64::new(0),
1234        })
1235    }
1236
1237    /// Read the events of the kernel and dispatch them, until the watcher
1238    /// got dropped or reading failed.
1239    async fn read_events(
1240        inotify: Arc<AsyncFd<InotifyFd>>,
1241        watches: Arc<Mutex<Watches>>,
1242        failed: Arc<AtomicBool>,
1243    ) {
1244        loop {
1245            let events = match inotify
1246                .async_io(Interest::READABLE, |fd| {
1247                    fd.0.read_events().map_err(io::Error::from)
1248                })
1249                .await
1250            {
1251                Ok(events) => events,
1252                Err(e) if e.kind() == ErrorKind::Interrupted => continue,
1253                Err(e) => {
1254                    error!("Unable to read watch events, checking on exit only: {e}");
1255                    break;
1256                }
1257            };
1258            let Ok(mut watches) = lock(&watches) else {
1259                break;
1260            };
1261            for event in &events {
1262                watches.dispatch(event);
1263            }
1264        }
1265        Self::stopped(&watches, &failed);
1266    }
1267
1268    /// The reader stopped. The next subscription gets a new watcher. The
1269    /// watches get dropped with their senders, so that the channels of the
1270    /// subscribers end once they took the events so far, they watch again
1271    /// then.
1272    fn stopped(watches: &Mutex<Watches>, failed: &AtomicBool) {
1273        failed.store(true, Ordering::Release);
1274        let watches = std::mem::take(&mut *watches.lock().unwrap_or_else(PoisonError::into_inner));
1275        drop(watches);
1276    }
1277
1278    /// Start watching `path` on behalf of `tx`. The watch gets removed when the
1279    /// returned subscription is dropped and no other subscriber of the watch
1280    /// is left.
1281    fn subscribe(self: &Arc<Self>, path: &Path, tx: Sender<WatchEvent>) -> Result<Subscription> {
1282        let id = self.next_id.fetch_add(1, Ordering::Relaxed);
1283        let mut watches = lock(&self.watches)?;
1284
1285        // inotify returns the existing descriptor for a file which is watched
1286        // already, the subscriber joins its watch then.
1287        let wd = self
1288            .inotify
1289            .get_ref()
1290            .0
1291            .add_watch(path, WATCH_FLAGS)
1292            .with_context(|| format!("watch {}", path.display()))?;
1293        watches
1294            .0
1295            .entry(wd)
1296            .or_insert_with(|| Watch {
1297                path: path.into(),
1298                subscribers: vec![],
1299            })
1300            .subscribers
1301            .push((id, tx));
1302
1303        Ok(Subscription {
1304            watcher: self.clone(),
1305            id,
1306            wd,
1307        })
1308    }
1309
1310    fn unsubscribe(&self, id: u64, wd: WatchDescriptor) {
1311        let Ok(mut watches) = self.watches.lock() else {
1312            return;
1313        };
1314        // The kernel removed the watch already.
1315        let Some(watch) = watches.0.get_mut(&wd) else {
1316            return;
1317        };
1318        watch.subscribers.retain(|(i, _)| *i != id);
1319        if !watch.subscribers.is_empty() {
1320            return;
1321        }
1322        let Some(watch) = watches.0.remove(&wd) else {
1323            return;
1324        };
1325        if let Err(e) = self.inotify.get_ref().0.rm_watch(wd) {
1326            // The watch is gone already if the cgroup got removed.
1327            trace!("Unable to unwatch {}: {:#}", watch.path.display(), e);
1328        }
1329    }
1330}

1374/// Reads the counter of the eventfd, which needs room for its 8 bytes.
1375impl AsyncRead for AsyncEventFd {
1376    fn poll_read(
1377        self: Pin<&mut Self>,
1378        cx: &mut TaskContext<'_>,
1379        buf: &mut ReadBuf<'_>,
1380    ) -> Poll<io::Result<()>> {
1381        if buf.remaining() < size_of::<u64>() {
1382            return Poll::Ready(Err(io::Error::from(io::ErrorKind::InvalidInput)));
1383        }
1384        loop {
1385            let mut guard = ready!(self.0.poll_read_ready(cx))?;
1386            match guard.try_io(|fd| fd.get_ref().read().map_err(io::Error::from)) {
1387                Ok(result) => {
1388                    return Poll::Ready(result.map(|count| buf.put_slice(&count.to_ne_bytes())));
1389                }
1390                Err(_would_block) => {}
1391            }
1392        }
1393    }

2097    /// stands in for the eventfd.
2098    struct FakeCgroupV1 {
2099        dir: tempfile::TempDir,
2100        event_tx: tokio::io::DuplexStream,
2101        event_rx: Option<tokio::io::DuplexStream>,
2102        token: CancellationToken,
2103    }
2104
2105    impl FakeCgroupV1 {
2106        fn new(oom_control: &str) -> Result<Self> {
2107            let dir = tempfile::tempdir()?;
2108            std::fs::write(dir.path().join("memory.oom_control"), oom_control)?;
2109            let (event_tx, event_rx) = tokio::io::duplex(64);
2110            Ok(Self {
2111                dir,
2112                event_tx,
2113                event_rx: Some(event_rx),
2114                token: CancellationToken::new(),
2115            })
2116        }
2117
2118        fn path(&self) -> PathBuf {
2119            self.dir.path().join("memory.oom_control")
2120        }
2121
2122        fn set(&self, oom_control: &str) -> Result<()> {
2123            Ok(std::fs::write(self.path(), oom_control)?)
2124        }
2125
2126        /// Signal the eventfd `count` times since the last read.
2127        async fn signal(&mut self, count: u64) -> Result<()> {
2128            Ok(self.event_tx.write_all(&count.to_ne_bytes()).await?)
2129        }
2130
2131        fn watch(&mut self, baseline: Option<u64>) -> task::JoinHandle<bool> {
2132            let watch = self.watch_event(baseline);
2133            task::spawn(async move { watch.await.is_ok_and(|event| event.oom) })
2134        }
2135
2136        fn watch_event(&mut self, baseline: Option<u64>) -> task::JoinHandle<OOMEvent> {
2137            let event_rx = self.event_rx.take().expect("watched once");
2138            let token = self.token.clone();
2139            let path = self.path();
2140            task::spawn(async move {
2141                OOMWatcher::watch_cgroup_v1(&token, event_rx, &path, baseline).await
2142            })
2143        }
2144    }
