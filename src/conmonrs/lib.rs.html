1//! A pod level OCI container runtime monitor for Linux.
2//!
3//! A container engine like CRI-O starts one `conmonrs` per pod and talks to
4//! it over a Cap'n Proto RPC interface on a UNIX socket, which this crate
5//! serves. The crate is the server only, the Go client lives in the same
6//! repository under `pkg/client`. The repository holds the full documentation
7//! for this version:
8#![doc = concat!(
9    "\n- <https://github.com/containers/conmon-rs/blob/v",
10    env!("CARGO_PKG_VERSION"),
11    "/README.md> on what conmon-rs does and how it is built.\n",
12    "- <https://github.com/containers/conmon-rs/blob/v",
13    env!("CARGO_PKG_VERSION"),
14    "/usage.md> on the command line, the RPCs and the attach, streaming and fd ",
15    "socket protocols.\n"
16)]

25/// Run conmonrs: print the version, run the `pause` subcommand or start the
26/// server, depending on the command line.
27pub fn run() -> Result<()> {
28    let config = Config::parse_args();
29
30    if let Some(v) = config.version() {
31        Version::new(v == Verbosity::Full).print();
32        return Ok(());
33    }
34
35    if let Some(v) = config.version_json() {
36        return Version::new(v == Verbosity::Full).print_json();
37    }
38
39    if let Some(Commands::Pause {
40        base_path,
41        pod_id,
42        ipc,
43        pid,
44        net,
45        user,
46        uts,
47        uid_mappings,
48        gid_mappings,
49    }) = config.command()
50    {
51        return Pause::run(
52            base_path,
53            pod_id,
54            *ipc,
55            *pid,
56            *net,
57            *user,
58            *uts,
59            uid_mappings,
60            gid_mappings,
61        )
62        .context("run pause");
63    }
64
65    config.validate().context("validate config")?;
66    config.prepare().context("prepare runtime directories")?;
67
68    Server::new(config)
69        .context("create server")?
70        .start()
71        .context("start server")
72}
