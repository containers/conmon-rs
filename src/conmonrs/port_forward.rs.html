1//! Port forwarding into the network namespace of a pod, the Kubernetes
2//! `portforward.k8s.io` protocol, as k8s.io/cri-streaming serves it for
3//! CRI-O otherwise.
4//!
5//! Over SPDY, the client opens two streams per forwarded connection, both
6//! with the `port` and `requestID` headers: an `error` stream, on which the
7//! server reports why forwarding failed, and a `data` stream, which carries
8//! the connection. The kubelet always uses SPDY, direct clients like crictl
9//! may tunnel it through a WebSocket instead.
10//!
11//! A request waits for its second stream without reading the first one.
12//! client-go opens the error stream first and waits for its reply, a client
13//! which sends a lot on a data stream before opening its error stream holds
14//! up the connection until the request times out.
15//!
16//! The older WebSocket protocol forwards one connection per port in the
17//! query of the request. Each port gets a data and an error channel, every
18//! message starts with its channel and the first one of each channel carries
19//! the port.
20

87/// How much the connections of a session which get established hold for the
88/// client together. Their data moves from the byte budget of the client into
89/// this one, so that connections to a port which does not answer, which the
90/// connect timeout could take long, do not stop the others. Once it is used
91/// up, connecting streams stop taking their data, which then stays in the
92/// byte budget of the client and may stop reading the client like any
93/// stream which does not take its data. Any number of connections get
94/// established at once, like with CRI-O.

98/// clients send per frame.

141/// A protocol a client may forward ports with over a WebSocket.
142#[derive(Clone, Copy, Debug, PartialEq)]
143pub enum WebSocketProtocol {
144    /// SPDY tunneled through binary messages.
145    SpdyTunnel,
146
147    /// The channel protocol, its payloads base64 encoded in text messages or
148    /// as they are in binary ones.
149    Channels { base64: bool },
150}
151
152impl WebSocketProtocol {
153    /// The protocol for the subprotocols the client requested, with the one
154    /// the server answers with. The tunnel wins like for the kubelet,
155    /// otherwise the first supported one, none means binary channels.
156    pub fn negotiate<'a>(
157        requested: impl IntoIterator<Item = &'a str>,
158    ) -> Option<(Self, Option<&'static str>)> {
159        let requested: Vec<_> = requested.into_iter().collect();
160        if requested.contains(&SPDY_TUNNEL_PROTOCOL) {
161            return Some((Self::SpdyTunnel, Some(SPDY_TUNNEL_PROTOCOL)));
162        }
163        if requested.is_empty() {
164            return Some((Self::Channels { base64: false }, None));
165        }
166        requested.into_iter().find_map(|protocol| match protocol {
167            CHANNEL_PROTOCOL => Some((Self::Channels { base64: false }, Some(CHANNEL_PROTOCOL))),
168            BASE64_CHANNEL_PROTOCOL => Some((
169                Self::Channels { base64: true },
170                Some(BASE64_CHANNEL_PROTOCOL),
171            )),
172            _ => None,
173        })
174    }
175}

234/// The streams of a request, until both arrived.
235struct PendingRequest {
236    data: Option<Stream>,
237    error: Option<Stream>,
238
239    /// The deadline of the request in [`Pending::deadlines`].
240    deadline: Key,
241}
242
243/// The requests waiting for their second stream, with their deadlines. Both
244/// are bounded by the streams of a connection.
245struct Pending {
246    requests: HashMap<Arc<str>, PendingRequest>,
247    deadlines: DelayQueue<Arc<str>>,
248}
249
250impl Pending {
251    fn new() -> Self {
252        Self {
253            requests: HashMap::new(),
254            deadlines: DelayQueue::new(),
255        }
256    }
257
258    /// The request, created with its deadline if it is new.
259    fn get_or_insert(&mut self, request_id: &Arc<str>) -> &mut PendingRequest {
260        let Self {
261            requests,
262            deadlines,
263        } = self;
264        requests
265            .entry(request_id.clone())
266            .or_insert_with(|| PendingRequest {
267                data: None,
268                error: None,
269                deadline: deadlines.insert(request_id.clone(), STREAM_CREATION_TIMEOUT),
270            })
271    }
272
273    /// Remove a request which got both streams, with its deadline.
274    fn complete(&mut self, request_id: &str) -> Option<PendingRequest> {
275        let request = self.requests.remove(request_id)?;
276        self.deadlines.remove(&request.deadline);
277        Some(request)
278    }
279}

290/// Creates the sockets of a session in its network namespace, which stay in
291/// the namespace they got created in. The runtime connects them. A thread
292/// enters the namespace and serves the connections which follow within the
293/// linger time, then it ends and leaves the namespace, at once when the
294/// session ends. Failures are not kept, every connection which fails gets its
295/// own error.
296struct Namespace {
297    path: PathBuf,
298    linger: Duration,
299
300    /// The requests of the thread which got started last.
301    requests: Mutex<Option<SocketRequests>>,
302
303    /// What the connections which get established hold, see
304    /// [`CONNECT_BUDGET`].
305    connect_budget: Budget,
306
307    /// How many threads got started and ended.
308    #[cfg(test)]
309    threads: AtomicUsize,
310    #[cfg(test)]
311    ended: Arc<AtomicUsize>,
312}
313
314impl Namespace {
315    fn new(path: PathBuf) -> Arc<Self> {
316        Self::with_linger(path, SOCKET_THREAD_LINGER)
317    }
318
319    fn with_linger(path: PathBuf, linger: Duration) -> Arc<Self> {
320        Arc::new(Self {
321            path,
322            linger,
323            requests: Mutex::default(),
324            connect_budget: Budget::new(CONNECT_BUDGET),
325            #[cfg(test)]
326            threads: AtomicUsize::new(0),
327            #[cfg(test)]
328            ended: Arc::default(),
329        })
330    }
331
332    async fn sockets(&self) -> Result<Sockets> {
333        // A thread which ended before it took the request gets replaced
334        // once.
335        for _ in 0..2 {
336            let (tx, rx) = oneshot::channel();
337            self.send(tx)?;
338            if let Ok(sockets) = rx.await {
339                return sockets;
340            }
341        }
342        bail!("socket thread ended")
343    }
344
345    /// Send a request to the thread, which gets started if it ended.
346    fn send(&self, reply: oneshot::Sender<Result<Sockets>>) -> Result<()> {
347        let running = self.lock().clone();
348        let reply = match running {
349            Some(requests) => match requests.send(reply) {
350                Ok(()) => return Ok(()),
351                Err(std_mpsc::SendError(reply)) => reply,
352            },
353            None => reply,
354        };
355        let requests = self.start()?;
356        requests
357            .send(reply)
358            .map_err(|_| anyhow::anyhow!("socket thread ended"))?;
359        *self.lock() = Some(requests);
360        Ok(())
361    }
362
363    fn lock(&self) -> MutexGuard<'_, Option<SocketRequests>> {
364        self.requests.lock().unwrap_or_else(PoisonError::into_inner)
365    }
366
367    fn start(&self) -> Result<SocketRequests> {
368        let (tx, rx) = std_mpsc::channel();
369        let (path, linger) = (self.path.clone(), self.linger);
370        #[cfg(test)]
371        let ended = self.ended.clone();
372        thread::Builder::new()
373            .name("portforward".into())
374            .stack_size(SOCKET_THREAD_STACK_SIZE)
375            .spawn(move || {
376                Self::serve(&path, linger, &rx);
377                #[cfg(test)]
378                ended.fetch_add(1, Ordering::Relaxed);
379            })
380            .context("spawn socket thread")?;
381        #[cfg(test)]
382        self.threads.fetch_add(1, Ordering::Relaxed);
383        Ok(tx)
384    }
385
386    /// Create sockets until no request arrived for the linger time or the
387    /// session is gone. The thread enters the namespace for its first
388    /// request, a request which finds it outside of the namespace tries again
389    /// and gets its own error.
390    fn serve(
391        path: &Path,
392        linger: Duration,
393        requests: &std_mpsc::Receiver<oneshot::Sender<Result<Sockets>>>,
394    ) {
395        let mut entered = false;
396        while let Ok(reply) = requests.recv_timeout(linger) {
397            if !entered {
398                if let Err(e) = Self::enter(path) {
399                    let _ = reply.send(Err(e));
400                    continue;
401                }
402                entered = true;
403            }
404            let _ = reply.send(Ok(Self::create_sockets()));
405        }
406    }
407
408    /// Enter the network namespace with the current thread.
409    fn enter(path: &Path) -> Result<()> {
410        let netns = File::open(path)
411            .with_context(|| format!("open network namespace {}", path.display()))?;
412        // Entering the own namespace needs privileges as well, which a server
413        // for a host network pod or in tests may not have.
414        let current = fs::metadata("/proc/thread-self/ns/net").context("stat network namespace")?;
415        let target = netns.metadata().context("stat target network namespace")?;
416        if (current.dev(), current.ino()) != (target.dev(), target.ino()) {
417            setns(&netns, CloneFlags::CLONE_NEWNET)
418                .with_context(|| format!("enter network namespace {}", path.display()))?;
419        }
420        Ok(())
421    }
422
423    fn create_sockets() -> Sockets {
424        [
425            (AddressFamily::Inet, IpAddr::V4(Ipv4Addr::LOCALHOST)),
426            (AddressFamily::Inet6, IpAddr::V6(Ipv6Addr::LOCALHOST)),
427        ]
428        .into_iter()
429        .map(|(family, ip)| {
430            let socket = socket(
431                family,
432                SockType::Stream,
433                SockFlag::SOCK_NONBLOCK | SockFlag::SOCK_CLOEXEC,
434                None,
435            )
436            .map(StdTcpStream::from)
437            .map_err(io::Error::from);
438            (ip, socket)
439        })
440        .collect()
441    }
442}

460    /// Serve port forward requests on the upgraded connection until it ends.
461    pub async fn serve_spdy<T>(self, io: T)
462    where
463        T: tokio::io::AsyncRead + tokio::io::AsyncWrite + Send + 'static,
464    {
465        let mut connection =
466            Connection::serve(io, self.idle_timeout, self.ping_interval, Self::accept);
467        let namespace = Namespace::new(self.netns_path);
468        let mut pending = Pending::new();
469
470        loop {
471            select! {
472                stream = connection.accept() => match stream {
473                    Some(stream) => Self::add(&mut pending, stream, &namespace),
474                    None => break,
475                },
476                Some(expired) = pending.deadlines.next() => {
477                    Self::expire(&mut pending, expired.into_inner());
478                }
479            }
480        }
481        debug!("Port forward connection ended");
482    }

484    /// The ports a channel protocol request forwards, by its query. The
485    /// `port` parameter may repeat and hold several ports separated by
486    /// commas.
487    pub fn query_ports(query: Option<&str>) -> Result<Vec<u16>> {
488        let mut ports = vec![];
489        for pair in query.unwrap_or_default().split('&') {
490            let (name, value) = pair.split_once('=').unwrap_or((pair, ""));
491            if Self::percent_decode(name) != PORT_QUERY {
492                continue;
493            }
494            let value = Self::percent_decode(value);
495            ensure!(
496                !value.is_empty(),
497                "query parameter {PORT_QUERY:?} cannot be empty"
498            );
499            for port in value.split(',') {
500                let port: u16 = port
501                    .parse()
502                    .with_context(|| format!("unable to parse {port:?} as a port"))?;
503                ensure!(port > 0, "port {port} must be > 0");
504                ports.push(port);
505            }
506        }
507        ensure!(
508            !ports.is_empty(),
509            "query parameter {PORT_QUERY:?} is required"
510        );
511        ensure!(
512            ports.len() <= MAX_CHANNEL_PORTS,
513            "at most {MAX_CHANNEL_PORTS} ports can be forwarded at once"
514        );
515        Ok(ports)
516    }

518    /// Decode a query component, invalid escapes stay as they are.
519    fn percent_decode(s: &str) -> String {
520        percent_decode_str(&s.replace('+', " "))
521            .decode_utf8_lossy()
522            .into_owned()
523    }

525    /// Serve port forwarding on the upgraded WebSocket until it ends. The
526    /// channel protocol forwards the ports, the tunnel ignores them.
527    pub async fn serve_websocket(
528        self,
529        socket: WebSocket,
530        protocol: WebSocketProtocol,
531        ports: Vec<u16>,
532    ) {
533        match protocol {
534            WebSocketProtocol::SpdyTunnel => {
535                let (server, tunnel) = tokio::io::duplex(TUNNEL_BUFFER);
536                let spdy = tokio::spawn(self.serve_spdy(server));
537                Self::tunnel(socket, tunnel).await;
538                let _ = spdy.await;
539            }
540            WebSocketProtocol::Channels { base64 } => {
541                self.serve_channels(socket, ports, base64).await
542            }
543        }
544        debug!("Port forward WebSocket ended");
545    }

547    /// Carry the bytes of the SPDY connection in binary messages, until
548    /// either side ends.
549    async fn tunnel(socket: WebSocket, connection: DuplexStream) {
550        let (mut sink, mut messages) = socket.split();
551        let (mut from_server, mut to_server) = tokio::io::split(connection);
552
553        let mut to_client: JoinHandle<()> = tokio::spawn(async move {
554            let mut buf = BytesMut::new();
555            loop {
556                buf.reserve(TUNNEL_BUFFER);
557                match from_server.read_buf(&mut buf).await {
558                    Ok(0) | Err(_) => break,
559                    Ok(_) => {
560                        if sink
561                            .send(Message::Binary(buf.split().freeze()))
562                            .await
563                            .is_err()
564                        {
565                            return;
566                        }
567                    }
568                }
569            }
570            let _ = sink
571                .send(Message::Close(Some(CloseFrame {
572                    code: CloseCode::Normal,
573                    reason: "".into(),
574                })))
575                .await;
576        });
577
578        let from_client = async {
579            while let Some(message) = messages.next().await {
580                match message {
581                    Ok(Message::Binary(data)) => {
582                        if to_server.write_all(&data).await.is_err() {
583                            break;
584                        }
585                    }
586                    Ok(Message::Text(_)) => {
587                        debug!("Ending port forward tunnel, the client sent a text message");
588                        break;
589                    }
590                    Ok(Message::Ping(_) | Message::Pong(_) | Message::Frame(_)) => {}
591                    Ok(Message::Close(_)) | Err(_) => break,
592                }
593            }
594            // Ends the SPDY connection, which ends the output as well.
595            let _ = to_server.shutdown().await;
596        };
597
598        let mut from_client = pin!(from_client);
599        let client_done = select! {
600            _ = &mut from_client => true,
601            // The SPDY connection ended, the client answers the close.
602            _ = &mut to_client => false,
603        };
604        if client_done {
605            Self::join_or_abort(to_client, WEBSOCKET_CLOSE_TIMEOUT).await;
606        } else {
607            let _ = time::timeout(WEBSOCKET_CLOSE_TIMEOUT, from_client).await;
608        }
609    }

611    /// Serve the channel protocol: forward a connection per port until all
612    /// of them ended, the client went away or the WebSocket was idle.
613    async fn serve_channels(self, socket: WebSocket, ports: Vec<u16>, base64: bool) {
614        let (mut sink, mut messages) = socket.split();
615        let (output_tx, mut output_rx) = mpsc::channel(OUTPUT_QUEUE);
616        let activity = Activity::new();
617
618        // On its own, as a port which does not take its input stops reading
619        // the WebSocket.
620        let idle = CancellationToken::new();
621        // Ends the watch however the session ends.
622        let _idle_guard = idle.clone().drop_guard();
623        tokio::spawn({
624            let (activity, idle, timeout) = (activity.clone(), idle.clone(), self.idle_timeout);
625            async move {
626                select! {
627                    _ = activity.idle(timeout) => {
628                        debug!("Closing idle port forward WebSocket");
629                        idle.cancel();
630                    }
631                    _ = idle.cancelled() => {}
632                }
633            }
634        });
635
636        let writer_activity = activity.clone();
637        let writer = tokio::spawn(async move {
638            while let Some(message) = output_rx.recv().await {
639                if sink.send(message).await.is_err() {
640                    return;
641                }
642                writer_activity.touch();
643            }
644            let _ = sink
645                .send(Message::Close(Some(CloseFrame {
646                    code: CloseCode::Normal,
647                    reason: "".into(),
648                })))
649                .await;
650        });
651
652        let namespace = Namespace::new(self.netns_path);
653        let budget = Budget::new(CHANNEL_BUFFER);
654        let mut inputs = Vec::with_capacity(ports.len());
655        let mut forwards = JoinSet::new();
656        for (i, port) in ports.into_iter().enumerate() {
657            let output = ChannelOutput {
658                messages: output_tx.clone(),
659                base64,
660                data: (i * 2) as u8,
661                error: (i * 2 + 1) as u8,
662            };
663            // The client learns the port of each channel from its first
664            // message.
665            let prefix = port.to_le_bytes();
666            if output.send(output.data, &prefix).await.is_err()
667                || output.send(output.error, &prefix).await.is_err()
668            {
669                return;
670            }
671            let (input_tx, input_rx) = mpsc::channel(CHANNEL_QUEUE);
672            inputs.push(input_tx);
673            forwards.spawn(Self::forward_channel(
674                namespace.clone(),
675                port,
676                input_rx,
677                output,
678            ));
679        }
680        drop(output_tx);
681
682        loop {
683            select! {
684                // Nothing gets dispatched after the idle timeout, and the
685                // session ends with its last forward, also while the client
686                // keeps sending.
687                biased;
688                _ = idle.cancelled() => break,
689                done = forwards.join_next() => if done.is_none() {
690                    break;
691                },
692                message = messages.next() => {
693                    // Pings and pongs only keep the WebSocket alive, the
694                    // heartbeat of a client does not keep it from going idle.
695                    let data = match message {
696                        Some(Ok(Message::Binary(data))) => data,
697                        Some(Ok(Message::Text(text))) => Bytes::from(text),
698                        Some(Ok(Message::Ping(_) | Message::Pong(_) | Message::Frame(_))) => continue,
699                        Some(Ok(Message::Close(_)) | Err(_)) | None => break,
700                    };
701                    activity.touch();
702                    Self::dispatch(&inputs, &budget, base64, data, &idle).await;
703                    // A dispatch which ended with the idle timeout is no
704                    // activity.
705                    if idle.is_cancelled() {
706                        break;
707                    }
708                    // A port which takes its input slowly keeps the WebSocket
709                    // active.
710                    activity.touch();
711                }
712            }
713        }
714
715        // The ports still forwarded lose their client.
716        forwards.abort_all();
717        while forwards.join_next().await.is_some() {}
718        Self::join_or_abort(writer, WEBSOCKET_CLOSE_TIMEOUT).await;
719    }

721    /// Wait for a task, abort it after the timeout. Dropping its handle would
722    /// leave it running, a writer to a client which does not read would hold
723    /// the connection.
724    async fn join_or_abort(mut task: JoinHandle<()>, timeout: Duration) {
725        if time::timeout(timeout, &mut task).await.is_err() {
726            task.abort();
727        }
728    }

730    /// Hand a message of the client to the connection of its channel. Only
731    /// the data channels, the even ones, take input. Waits while a connected
732    /// port does not take its input or the ports hold all of the budget,
733    /// which stops reading the WebSocket like the reference implementation
734    /// does, the protocol has no flow control. Connecting ports take it, see
735    /// [`Self::drain_until`], and the idle timeout still ends the wait.
736    async fn dispatch(
737        inputs: &[mpsc::Sender<Held>],
738        budget: &Budget,
739        base64: bool,
740        message: Bytes,
741        idle: &CancellationToken,
742    ) {
743        let Some((channel, data)) = channel::decode(base64, message) else {
744            return;
745        };
746        let Some(input) = inputs
747            .get(usize::from(channel / 2))
748            .filter(|_| channel % 2 == 0)
749        else {
750            trace!("Dropping port forward message for channel {channel}");
751            return;
752        };
753        if !data.is_empty() {
754            select! {
755                // A connection which ended does not take input anymore.
756                _ = async { input.send(budget.hold(data).await).await } => {}
757                // Also while waiting for the budget.
758                _ = input.closed() => {}
759                _ = idle.cancelled() => {}
760            }
761        }
762    }

764    /// Forward the connection of a port, the error channel reports a failure.
765    async fn forward_channel(
766        namespace: Arc<Namespace>,
767        port: u16,
768        mut input: mpsc::Receiver<Held>,
769        output: ChannelOutput,
770    ) {
771        debug!("Forwarding port {port} over WebSocket");
772        let result = match Self::establish(&namespace, port, &mut input).await {
773            Ok((connection, early)) => {
774                Self::copy_channel(early, &mut input, &output, connection).await
775            }
776            Err(e) => Err(e),
777        };
778        if let Err(e) = result {
779            let message = format!("error forwarding port {port}: {e:#}");
780            warn!("Port forward over WebSocket: {message}");
781            let _ = output.send(output.error, message.as_bytes()).await;
782        }
783    }

785    /// Copy between the channels of a port and its connection in the pod,
786    /// until the pod closed the connection or the client went away. The
787    /// protocol has no way to close a channel, the input after the pod closed
788    /// the connection does not matter.
789    async fn copy_channel(
790        early: Early,
791        input: &mut mpsc::Receiver<Held>,
792        output: &ChannelOutput,
793        connection: TcpStream,
794    ) -> Result<()> {
795        let (mut from_container, mut to_container) = connection.into_split();
796        let upstream = Self::write_input(early, input, &mut to_container);
797
798        let downstream = async {
799            let mut buf = vec![0; READ_BUFFER];
800            loop {
801                let read = from_container
802                    .read(&mut buf)
803                    .await
804                    .context("read from the pod")?;
805                if read == 0 {
806                    return anyhow::Ok(());
807                }
808                if output.send(output.data, &buf[..read]).await.is_err() {
809                    // The client went away.
810                    return Ok(());
811                }
812            }
813        };
814
815        select! {
816            result = downstream => result,
817            result = upstream => result,
818        }
819    }

822    fn accept(headers: &Headers) -> Result<()> {
823        Self::port(headers)?;

825            Some(STREAM_TYPE_DATA | STREAM_TYPE_ERROR) => Ok(()),
826            Some(typ) => bail!("invalid stream type {typ:?}"),
827            None => bail!("{STREAM_TYPE_HEADER:?} header is required"),

832        let port = headers
833            .get(PORT_HEADER)

846    /// The request a stream belongs to. Old clients send no ID, their
847    /// requests use the IDs of the error streams, the data streams follow
848    /// them.
849    fn request_id(stream: &Stream) -> Arc<str> {
850        match stream.headers().get(REQUEST_ID_HEADER) {
851            Some(id) if !id.is_empty() => id.into(),
852            _ if Self::is_error_stream(stream) => stream.id().to_string().into(),
853            _ => (i64::from(stream.id()) - 2).to_string().into(),
854        }
855    }

857    /// Add a stream to its request, which starts once both streams arrived.
858    fn add(pending: &mut Pending, stream: Stream, namespace: &Arc<Namespace>) {
859        let request_id = Self::request_id(&stream);
860        let request = pending.get_or_insert(&request_id);
861        let slot = if Self::is_error_stream(&stream) {
862            &mut request.error
863        } else {
864            &mut request.data
865        };
866        if slot.is_some() {
867            // Dropping it resets it.
868            warn!(
869                "Rejecting stream {} of port forward request {request_id}, the request has one of its type already",
870                stream.id()
871            );
872            return;
873        }
874        *slot = Some(stream);
875
876        if request.data.is_none() || request.error.is_none() {
877            return;
878        }
879        if let Some(PendingRequest {
880            data: Some(data),
881            error: Some(error),
882            ..
883        }) = pending.complete(&request_id)
884        {
885            tokio::spawn(Self::forward(request_id, data, error, namespace.clone()));
886        }
887    }

889    /// Fail a request whose second stream did not arrive in time.
890    fn expire(pending: &mut Pending, request_id: Arc<str>) {
891        let Some(mut request) = pending.requests.remove(&request_id) else {
892            return;
893        };
894        warn!("Port forward request {request_id} timed out waiting for streams");
895        if let Some(mut error) = request.error.take() {
896            let message = format!("request {request_id}: timed out waiting for streams");
897            tokio::spawn(async move {
898                let _ = error.send(message.into()).await;
899                let _ = error.close().await;
900            });
901        }
902    }

904    /// Forward a single connection, the error stream reports a failure.
905    async fn forward(
906        request_id: Arc<str>,
907        data: Stream,
908        mut error: Stream,
909        namespace: Arc<Namespace>,
910    ) {
911        let port = Self::port(data.headers()).unwrap_or_default();
912        debug!("Forwarding port {port} for request {request_id}");
913
914        let (mut from_client, to_client) = data.split();
915        let result = match Self::establish(&namespace, port, &mut from_client).await {
916            Ok((connection, early)) => Self::copy(early, from_client, to_client, connection).await,
917            Err(e) => {
918                // Dropping the sending side resets the data stream.
919                drop(to_client);
920                Err(e)
921            }
922        };
923        match result {
924            Ok(()) => debug!("Done forwarding port {port} for request {request_id}"),
925            Err(e) => {
926                let message = format!("error forwarding port {port}: {e:#}");
927                warn!("Port forward request {request_id}: {message}");
928                if let Err(e) = error.send(message.into()).await {
929                    debug!("Unable to report port forward error: {e:#}");
930                }
931            }
932        }
933        if let Err(e) = error.close().await {
934            debug!("Unable to close port forward error stream: {e:#}");
935        }
936    }

939    /// meanwhile.
940    async fn establish(
941        namespace: &Namespace,
942        port: u16,
943        input: &mut impl Input,
944    ) -> Result<(TcpStream, Early)> {

948            input,

953    /// Wait for the connection to the pod while taking what the client sends
954    /// meanwhile, which may take until the connect timeout. The data moves
955    /// from the byte budget of the client into the connect budget, so that
956    /// the client can go on with its other connections. Without room in the
957    /// connect budget no more data gets taken, see [`CONNECT_BUDGET`]. A
958    /// client which sends more than [`CONNECT_BUFFER`] fails the forwarding.
959    async fn drain_until<T>(
960        connect: impl Future<Output = Result<T>>,
961        connect_budget: &Budget,
962        input: &mut impl Input,
963    ) -> Result<(T, Early)> {
964        let mut connect = pin!(connect);
965        let mut early = Early::default();
966        let mut len = 0;
967        // Data which got taken and waits for room in the connect budget.
968        let mut taken: Option<Held> = None;
969        loop {
970            select! {
971                result = &mut connect => {
972                    early.data.extend(taken);
973                    return Ok((result?, early));
974                }
975                held = async {
976                    match &taken {
977                        Some(data) => connect_budget.hold(Bytes::clone(data)).await,
978                        None => future::pending().await,
979                    }
980                } => {
981                    // The data leaves the byte budget of the client.
982                    taken = None;
983                    early.data.push_back(held);
984                }
985                data = input.recv(), if taken.is_none() && !early.ended => match data {
986                    Some(data) => {
987                        len += data.len();
988                        ensure!(
989                            len <= CONNECT_BUFFER,
990                            "the client sent more than {CONNECT_BUFFER} bytes before the connection got established"
991                        );
992                        taken = Some(data);
993                    }
994                    None => early.ended = true,
995                },
996            }
997        }
998    }

1000    /// Copy between the data stream and the connection in the pod, until
1001    /// both sides closed or either reset. A failure resets the data stream.
1002    async fn copy(
1003        early: Early,
1004        mut from_client: StreamReceiver,
1005        mut to_client: StreamSender,
1006        connection: TcpStream,
1007    ) -> Result<()> {
1008        let (mut from_container, mut to_container) = connection.into_split();
1009
1010        let upstream = async {
1011            Self::write_input(early, &mut from_client, &mut to_container).await?;
1012            if !from_client.is_reset() {
1013                to_container
1014                    .shutdown()
1015                    .await
1016                    .context("close the connection to the pod")?;
1017            }
1018            anyhow::Ok(())
1019        };
1020
1021        let downstream = async {
1022            let mut buf = BytesMut::new();
1023            loop {
1024                buf.reserve(READ_BUFFER);
1025                let mut limited = (&mut buf).limit(READ_BUFFER);
1026                let read = select! {
1027                    read = from_container.read_buf(&mut limited) => read.context("read from the pod")?,
1028                    // Nothing takes the output anymore.
1029                    _ = to_client.reset_by_peer() => return Ok(()),
1030                };
1031                if read == 0 {
1032                    break;
1033                }
1034                let data: Bytes = buf.split().freeze();
1035                if let Err(e) = to_client.send(data).await {
1036                    return Self::unless_reset(&to_client, e);
1037                }
1038            }
1039            match to_client.close().await {
1040                Ok(()) => Ok(()),
1041                Err(e) => Self::unless_reset(&to_client, e),
1042            }
1043        };
1044
1045        let mut upstream = pin!(upstream);
1046        let mut downstream = pin!(downstream);
1047        select! {
1048            result = &mut downstream => {
1049                result?;
1050                if time::timeout(CLOSE_GRACE_PERIOD, upstream).await.is_err() {
1051                    debug!("Client did not close its side of the connection");
1052                }
1053                Ok(())
1054            }
1055            // The client closed its side, the pod may still answer.
1056            result = &mut upstream => {
1057                result?;
1058                downstream.await
1059            }
1060        }
1061    }

1063    /// A stream the client reset or whose connection ended is no failure.
1064    fn unless_reset(sender: &StreamSender, e: anyhow::Error) -> Result<()> {
1065        if sender.is_reset() { Ok(()) } else { Err(e) }
1066    }

1068    /// Write what the client sends to the pod until the client is done,
1069    /// starting with what it sent while the connection got established.
1070    async fn write_input(
1071        early: Early,
1072        input: &mut impl Input,
1073        to_container: &mut OwnedWriteHalf,
1074    ) -> Result<()> {
1075        for data in early.data {
1076            to_container
1077                .write_all(&data)
1078                .await
1079                .context("write to the pod")?;
1080        }
1081        if !early.ended {
1082            while let Some(data) = input.recv().await {
1083                to_container
1084                    .write_all(&data)
1085                    .await
1086                    .context("write to the pod")?;
1087            }
1088        }
1089        Ok(())
1090    }

1092    /// Connect to the port in the network namespace of the pod. The pod may
1093    /// listen on either family, they get tried one after the other like CRI-O
1094    /// does.
1095    async fn connect(namespace: &Namespace, port: u16) -> Result<TcpStream> {
1096        let mut errors = vec![];
1097        for (ip, socket) in namespace.sockets().await? {
1098            let result = match socket {
1099                Ok(socket) => time::timeout(
1100                    CONNECT_TIMEOUT,
1101                    TcpSocket::from_std_stream(socket).connect((ip, port).into()),
1102                )
1103                .await
1104                .unwrap_or_else(|_| Err(io::ErrorKind::TimedOut.into())),
1105                Err(e) => Err(e),
1106            };
1107            match result {
1108                Ok(connection) => return Ok(connection),
1109                Err(e) => errors.push(format!("{ip}: {e}")),
1110            }
1111        }
1112        bail!(
1113            "failed to connect to localhost:{port} inside network namespace {}: {}",
1114            namespace.path.display(),
1115            errors.join(", ")
1116        )
1117    }
