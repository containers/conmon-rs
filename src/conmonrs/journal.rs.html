30/// The syslog priorities used for the `PRIORITY` field.
31#[derive(Clone, Copy, Debug, Eq, PartialEq)]
32pub enum Priority {
33    Error = 3,
34    Warning = 4,
35    Notice = 5,
36    Info = 6,
37    Debug = 7,
38}
39
40/// A journal entry, encoded for the native protocol.
41#[derive(Debug, Default)]
42pub struct Entry(Vec<u8>);
43
44impl Entry {
45    /// An entry with the `PRIORITY` and `MESSAGE` fields.
46    pub fn new(priority: Priority, message: impl AsRef<[u8]>) -> Self {
47        let mut entry = Self::default();
48        entry.reset(priority, message);
49        entry
50    }
51
52    /// Start over with the `PRIORITY` and `MESSAGE` fields, keeping the
53    /// allocation.
54    pub fn reset(&mut self, priority: Priority, message: impl AsRef<[u8]>) -> &mut Self {
55        self.0.clear();
56        self.field("PRIORITY", [b'0' + priority as u8])
57            .field("MESSAGE", message)
58    }
59
60    /// Append a field. The key must be a valid journal field name: upper case
61    /// letters, digits and underscores, not starting with a digit or an
62    /// underscore.
63    pub fn field(&mut self, key: &str, value: impl AsRef<[u8]>) -> &mut Self {
64        debug_assert!(Self::valid_key(key), "invalid journal field name {key}");
65        let value = value.as_ref();
66        self.0.extend_from_slice(key.as_bytes());
67        // `KEY=value\n` cannot carry a newline in the value, those values get
68        // their length as 64 bit little endian integer instead.
69        if value.contains(&b'\n') {
70            self.0.push(b'\n');
71            self.0
72                .extend_from_slice(&u64::try_from(value.len()).unwrap_or(u64::MAX).to_le_bytes());
73        } else {
74            self.0.push(b'=');
75        }
76        self.0.extend_from_slice(value);
77        self.0.push(b'\n');
78        self
79    }
80
81    fn valid_key(key: &str) -> bool {
82        (1..=64).contains(&key.len())
83            && !key.starts_with(|c: char| c == '_' || c.is_ascii_digit())
84            && key
85                .chars()
86                .all(|c| c.is_ascii_uppercase() || c.is_ascii_digit() || c == '_')
87    }
88
89    /// Append the already encoded fields of another entry.
90    pub fn extend(&mut self, fields: &Entry) -> &mut Self {
91        self.0.extend_from_slice(&fields.0);
92        self
93    }
94
95    pub fn as_bytes(&self) -> &[u8] {
96        &self.0
97    }
98}

100/// How sending waits for a journal which does not keep up, for example while
101/// it restarts or its disk is slow. The modes use separate sockets.
102#[derive(Clone, Copy, Debug, Eq, PartialEq)]
103pub enum Wait {
104    /// Never block, the entry fails with `ErrorKind::WouldBlock`. For the
105    /// server logs, which tracing writes from the runtime threads.
106    Never,
107    /// Block for at most `SEND_TIMEOUT`, then fail with
108    /// `ErrorKind::TimedOut`. For the container logs, which get sent from
109    /// the blocking pool and should not be dropped on a short stall.
110    Bounded,
111}
112
113/// The longest `Wait::Bounded` blocks for one entry.
114const SEND_TIMEOUT: Duration = Duration::from_secs(1);
115
116/// The send buffer systemd uses for its journal clients, so that bursts and
117/// entries up to that size do not need a memfd.
118const SEND_BUFFER_SIZE: usize = 8 * 1024 * 1024;
119
120/// Send an entry to the journal socket at a path.
121pub fn send_at(wait: Wait, path: &Path, entry: &Entry) -> io::Result<()> {
122    match send_to(socket(wait)?, path, entry.as_bytes()) {
123        Err(e) if wait == Wait::Bounded && e.kind() == ErrorKind::WouldBlock => {
124            Err(io::Error::new(
125                ErrorKind::TimedOut,
126                format!("journal did not take the entry within {SEND_TIMEOUT:?}"),
127            ))
128        }
129        result => result,
130    }
131}
132
133/// The socket of a mode, created on first use. A failed creation is not
134/// stored, the next entry tries again.
135fn socket(wait: Wait) -> io::Result<&'static UnixDatagram> {
136    static NEVER: OnceLock<UnixDatagram> = OnceLock::new();
137    static BOUNDED: OnceLock<UnixDatagram> = OnceLock::new();
138    let cell = match wait {
139        Wait::Never => &NEVER,
140        Wait::Bounded => &BOUNDED,
141    };
142    if let Some(socket) = cell.get() {
143        return Ok(socket);
144    }
145    let socket = UnixDatagram::unbound()?;
146    match wait {
147        Wait::Never => socket.set_nonblocking(true)?,
148        Wait::Bounded => socket.set_write_timeout(Some(SEND_TIMEOUT))?,
149    }
150    // Best effort like systemd: forcing needs CAP_NET_ADMIN, without it the
151    // kernel caps the size at net.core.wmem_max.
152    if setsockopt(&socket, sockopt::SndBufForce, &SEND_BUFFER_SIZE).is_err() {
153        setsockopt(&socket, sockopt::SndBuf, &SEND_BUFFER_SIZE).ok();
154    }
155    Ok(cell.get_or_init(|| socket))
156}

158/// Send the data as one datagram, or as a memfd if it does not fit.
159fn send_to(socket: &UnixDatagram, path: &Path, data: &[u8]) -> io::Result<()> {
160    match socket.send_to(data, path) {
161        Ok(_) => Ok(()),
162        Err(e)
163            if matches!(
164                e.raw_os_error().map(Errno::from_raw),
165                Some(Errno::EMSGSIZE | Errno::ENOBUFS)
166            ) =>
167        {
168            send_memfd(socket, path, data)
169        }
170        Err(e) => Err(e),
171    }
172}

174/// Whether an error means that there is no journal to send to: no socket
175/// file, or no one listening on it.
176pub fn unavailable(error: &io::Error) -> bool {
177    matches!(
178        error.kind(),
179        ErrorKind::NotFound | ErrorKind::ConnectionRefused
180    )
181}

183/// Send the data in a sealed memfd, the only kind journald accepts, as
184/// `SCM_RIGHTS` of an otherwise empty datagram.
185fn send_memfd(socket: &UnixDatagram, path: &Path, data: &[u8]) -> io::Result<()> {
186    let mut file = File::from(create_memfd()?);
187    file.write_all(data)?;
188    fcntl(
189        &file,
190        FcntlArg::F_ADD_SEALS(
191            SealFlag::F_SEAL_SHRINK
192                | SealFlag::F_SEAL_GROW
193                | SealFlag::F_SEAL_WRITE
194                | SealFlag::F_SEAL_SEAL,
195        ),
196    )?;
197    sendmsg(
198        socket.as_raw_fd(),
199        &[],
200        &[ControlMessage::ScmRights(&[file.as_raw_fd()])],
201        MsgFlags::empty(),
202        Some(&UnixAddr::new(path)?),
203    )?;
204    Ok(())
205}
206
207/// Create a memfd which allows sealing. Like systemd, ask for a
208/// non-executable one first, kernels before 6.3 reject that flag.
209fn create_memfd() -> nix::Result<OwnedFd> {
210    let flags = MFdFlags::MFD_CLOEXEC | MFdFlags::MFD_ALLOW_SEALING;
211    let noexec = MFdFlags::from_bits_retain(libc::MFD_NOEXEC_SEAL);
212    match memfd_create(c"conmonrs-journal", flags | noexec) {
213        Err(Errno::EINVAL) => memfd_create(c"conmonrs-journal", flags),
214        result => result,
215    }
216}

218/// Writes every log line as one journal entry with the priority derived from
219/// the tracing level.
220#[derive(Default)]
221pub struct Journal {
222    priority: Option<Priority>,
223}
224
225impl Journal {
226    const DEFAULT_PRIORITY: Priority = Priority::Notice;
227
228    /// The entry of a formatted event. The journal separates entries itself,
229    /// so the line ending of the event is dropped.
230    fn entry(&self, line: &[u8]) -> Entry {
231        Entry::new(
232            self.priority.unwrap_or(Self::DEFAULT_PRIORITY),
233            line.strip_suffix(b"\n").unwrap_or(line),
234        )
235    }
236
237    fn priority_for(level: &Level) -> Priority {
238        match *level {
239            Level::ERROR => Priority::Error,
240            Level::WARN => Priority::Warning,
241            Level::INFO => Priority::Info,
242            Level::DEBUG | Level::TRACE => Priority::Debug,
243        }
244    }
245}
246
247impl Journal {
248    /// Send without blocking the runtime thread of the event. A system
249    /// without journald is not an error, like with `sd_journal_send`, and a
250    /// journal which does not keep up loses the line, otherwise every event
251    /// would report an error.
252    fn send_at(&self, path: &Path, line: &[u8]) -> io::Result<()> {
253        match send_at(Wait::Never, path, &self.entry(line)) {
254            Err(e) if unavailable(&e) || e.kind() == ErrorKind::WouldBlock => Ok(()),
255            result => result,
256        }
257    }
258}
259
260impl Write for Journal {
261    fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
262        self.send_at(Path::new(SOCKET_PATH), buf)?;
263        Ok(buf.len())
264    }
265
266    fn flush(&mut self) -> io::Result<()> {
267        Ok(())
268    }
269}
270
271impl<'a> MakeWriter<'a> for Journal {
272    type Writer = Journal;
273
274    fn make_writer(&'a self) -> Self::Writer {
275        Journal::default()
276    }
277
278    fn make_writer_for(&'a self, meta: &Metadata<'_>) -> Self::Writer {
279        Journal {
280            priority: Some(Self::priority_for(meta.level())),
281        }
282    }
283}

289    /// Decode the fields of an entry, the way journald does.
290    pub fn decode(mut data: &[u8]) -> Vec<(String, Vec<u8>)> {
291        let mut fields = vec![];
292        while !data.is_empty() {
293            let end = data.iter().position(|b| *b == b'\n').expect("newline");
294            let line = &data[..end];
295            if let Some(eq) = line.iter().position(|b| *b == b'=') {
296                fields.push((
297                    String::from_utf8(line[..eq].to_vec()).expect("utf8 key"),
298                    line[eq + 1..].to_vec(),
299                ));
300                data = &data[end + 1..];
301            } else {
302                let rest = &data[end + 1..];
303                let len = u64::from_le_bytes(rest[..8].try_into().expect("length")) as usize;
304                assert_eq!(rest[8 + len], b'\n', "value ends with a newline");
305                fields.push((
306                    String::from_utf8(line.to_vec()).expect("utf8 key"),
307                    rest[8..8 + len].to_vec(),
308                ));
309                data = &rest[8 + len + 1..];
310            }
311        }
312        fields
313    }

413    /// Receive one datagram, with its content or the content of the memfd it
414    /// carries.
415    fn receive(receiver: &UnixDatagram) -> anyhow::Result<(Vec<u8>, bool)> {
416        use nix::sys::socket::{ControlMessageOwned, recvmsg};
417        use std::{
418            io::{IoSliceMut, Read, Seek},
419            os::fd::{FromRawFd, RawFd},
420        };
421
422        let mut buf = vec![0; 4096];
423        let mut iov = [IoSliceMut::new(&mut buf)];
424        let mut cmsg = nix::cmsg_space!([RawFd; 1]);
425        let msg = recvmsg::<()>(
426            receiver.as_raw_fd(),
427            &mut iov,
428            Some(&mut cmsg),
429            MsgFlags::MSG_CMSG_CLOEXEC,
430        )?;
431        let len = msg.bytes;
432        let Some(ControlMessageOwned::ScmRights(fds)) = msg.cmsgs()?.next() else {
433            buf.truncate(len);
434            return Ok((buf, false));
435        };
436        #[allow(unsafe_code)]
437        // SAFETY: SCM_RIGHTS installed a new descriptor which nothing else owns.
438        let mut file = unsafe { File::from_raw_fd(fds[0]) };
439        let seals = fcntl(&file, FcntlArg::F_GET_SEALS)?;
440        assert!(
441            SealFlag::from_bits_truncate(seals)
442                .contains(SealFlag::F_SEAL_SHRINK | SealFlag::F_SEAL_GROW | SealFlag::F_SEAL_WRITE)
443        );
444        // The descriptor shares the offset with the one of the sender, which
445        // is at the end. journald maps the memfd instead.
446        file.rewind()?;
447        let mut data = vec![];
448        file.read_to_end(&mut data)?;
449        Ok((data, true))
450    }

452    /// Entries go inline if they fit into a datagram, otherwise as memfd.
453    #[test]
454    fn send_inline_and_as_memfd() -> anyhow::Result<()> {
455        use nix::sys::socket::{setsockopt, sockopt};
456
457        let dir = tempfile::tempdir()?;
458        let path = dir.path().join("socket");
459        let receiver = UnixDatagram::bind(&path)?;
460        let sender = UnixDatagram::unbound()?;
461        // A small send buffer makes the large entry fail inline, whatever
462        // the default of the host is.
463        setsockopt(&sender, sockopt::SndBuf, &4096)?;
464
465        let small = Entry::new(Priority::Info, "small");
466        send_to(&sender, &path, small.as_bytes())?;
467        assert_eq!(receive(&receiver)?, (small.as_bytes().to_vec(), false));
468
469        let large = Entry::new(Priority::Info, "x".repeat(1 << 20));
470        send_to(&sender, &path, large.as_bytes())?;
471        let (data, memfd) = receive(&receiver)?;
472        assert!(memfd);
473        assert!(data == large.as_bytes(), "memfd holds the entry");
474        Ok(())

477    /// Send small entries to a socket which is never read, until the queue
478    /// of the receiver is full and sending fails.
479    fn fill(wait: Wait, path: &Path) -> io::Error {
480        let entry = Entry::new(Priority::Info, "fill");
481        for _ in 0..100_000 {
482            if let Err(e) = send_at(wait, path, &entry) {
483                return e;
484            }
485        }
486        panic!("the receiver queue never filled up");
487    }
