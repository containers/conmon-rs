1//! File descriptor mapping for child process spawning.
2//!
3//! This mirrors the API of the `command-fds` crate (`CommandFdExt`,
4//! `FdMapping`, `FdMappingCollision`) without pulling in the dependency.
5
6use nix::{
7    fcntl::{FcntlArg, FdFlag, fcntl},
8    unistd::dup2_raw,
9};
10use std::{
11    cmp::max,
12    fmt, io,
13    os::fd::{AsRawFd, FromRawFd, IntoRawFd, OwnedFd, RawFd},
14};
15use tokio::process::Command;
16
17/// A mapping from a file descriptor in the parent to a file descriptor in the child.
18#[derive(Debug)]
19pub struct FdMapping {
20    pub parent_fd: OwnedFd,
21    pub child_fd: RawFd,
22}
23
24/// Extension to add file descriptor mappings to a Command.
25pub trait CommandFdExt {
26    fn fd_mappings(&mut self, mappings: Vec<FdMapping>) -> Result<&mut Self, FdMappingCollision>;
27}
28
29/// Error when two or more mappings target the same child FD.
30#[derive(Copy, Clone, Debug, Eq, PartialEq)]
31pub struct FdMappingCollision;
32
33impl fmt::Display for FdMappingCollision {
34    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
35        write!(f, "Two or more mappings for the same child FD")
36    }
37}
38
39impl std::error::Error for FdMappingCollision {}
40
41impl CommandFdExt for Command {
42    fn fd_mappings(
43        &mut self,
44        mut mappings: Vec<FdMapping>,
45    ) -> Result<&mut Self, FdMappingCollision> {
46        let child_fds = validate_child_fds(&mappings)?;
47
48        // SAFETY: map_fds will not allocate, so it is safe to call from this hook.
49        unsafe {
50            self.pre_exec(move || map_fds(&mut mappings, &child_fds));
51        }
52
53        Ok(self)
54    }
55}
56
57fn validate_child_fds(mappings: &[FdMapping]) -> Result<Vec<RawFd>, FdMappingCollision> {
58    let mut child_fds: Vec<RawFd> = mappings.iter().map(|m| m.child_fd).collect();
59    child_fds.sort_unstable();
60    child_fds.dedup();
61    if child_fds.len() != mappings.len() {
62        return Err(FdMappingCollision);
63    }
64    Ok(child_fds)
65}
66
67// This function must not allocate, as it is called from the pre_exec hook.
68fn map_fds(mappings: &mut [FdMapping], child_fds: &[RawFd]) -> io::Result<()> {
69    if mappings.is_empty() {
70        return Ok(());
71    }
72
73    let first_safe_fd = mappings
74        .iter()
75        .map(|m| max(m.parent_fd.as_raw_fd(), m.child_fd))
76        .max()
77        .expect("mappings is non-empty")
78        + 1;
79
80    for mapping in mappings.iter_mut() {
81        if child_fds.contains(&mapping.parent_fd.as_raw_fd())
82            && mapping.parent_fd.as_raw_fd() != mapping.child_fd
83        {
84            let parent_fd = fcntl(&mapping.parent_fd, FcntlArg::F_DUPFD_CLOEXEC(first_safe_fd))?;
85            // SAFETY: the duplicate was just created and nothing else owns it,
86            // the old descriptor gets closed when the mapping is overwritten.
87            unsafe {
88                mapping.parent_fd = OwnedFd::from_raw_fd(parent_fd);
89            }
90        }
91    }
92
93    for mapping in mappings {
94        if mapping.child_fd == mapping.parent_fd.as_raw_fd() {
95            fcntl(&mapping.parent_fd, FcntlArg::F_SETFD(FdFlag::empty()))?;
96        } else {
97            // SAFETY: the child descriptor number is not owned by anything
98            // else, the descriptors below first_safe_fd are the ones mapped
99            // here, and the result is leaked on purpose for the exec.
100            unsafe {
101                let _ = dup2_raw(&mapping.parent_fd, mapping.child_fd)?.into_raw_fd();
102            }
103        }
104    }
105
106    Ok(())
107}

109#[cfg(test)]
110mod tests {
111    use super::*;
112    use std::fs::File;
113
114    fn mapping(child_fd: RawFd) -> Result<FdMapping, std::io::Error> {
115        Ok(FdMapping {
116            parent_fd: File::open("/dev/null")?.into(),
117            child_fd,
118        })
119    }
120
121    #[test]
122    fn child_fds_must_be_unique() -> Result<(), std::io::Error> {
123        let mappings = [mapping(3)?, mapping(4)?];
124        assert_eq!(validate_child_fds(&mappings).unwrap(), vec![3, 4]);
125
126        let mappings = [mapping(3)?, mapping(3)?];
127        assert!(validate_child_fds(&mappings).is_err());
128        Ok(())
129    }
130
131    /// A descriptor which is the target of another mapping must be moved out
132    /// of the way before it gets overwritten.
133    #[tokio::test]
134    async fn swapped_fds_are_mapped() -> anyhow::Result<()> {
135        let dir = tempfile::tempdir()?;
136        let (a_path, b_path) = (dir.path().join("a"), dir.path().join("b"));
137        // Free numbers well above what the test harness uses.
138        let a: OwnedFd = File::create(&a_path)?.into();
139        let a = fcntl(&a, FcntlArg::F_DUPFD_CLOEXEC(200))?;
140        // SAFETY: the duplicate is new and owned by nothing else.
141        let a = unsafe { OwnedFd::from_raw_fd(a) };
142        let b: OwnedFd = File::create(&b_path)?.into();
143        let b = fcntl(&b, FcntlArg::F_DUPFD_CLOEXEC(200))?;
144        // SAFETY: as above.
145        let b = unsafe { OwnedFd::from_raw_fd(b) };
146        let (a_fd, b_fd) = (a.as_raw_fd(), b.as_raw_fd());
147
148        let output = Command::new("sh")
149            .args([
150                "-c",
151                &format!("readlink /proc/self/fd/{a_fd} /proc/self/fd/{b_fd}"),
152            ])
153            .fd_mappings(vec![
154                FdMapping {
155                    parent_fd: a,
156                    child_fd: b_fd,
157                },
158                FdMapping {
159                    parent_fd: b,
160                    child_fd: a_fd,
161                },
162            ])?
163            .output()
164            .await?;
165        assert!(output.status.success());
166        let stdout = String::from_utf8(output.stdout)?;
167        let links: Vec<_> = stdout.lines().collect();
168        assert_eq!(
169            links,
170            [b_path.to_str(), a_path.to_str()].map(Option::unwrap)
171        );
172        Ok(())
173    }
