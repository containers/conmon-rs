1//! The SPDY/3.1 frames and their encoding, see
2//! <https://www.chromium.org/spdy/spdy-protocol/spdy-protocol-draft3-1/>.
3//!
4//! Header blocks are compressed with zlib and a preset dictionary. Each
5//! direction of a connection keeps one compression context for all its header
6//! blocks, so every frame which carries one has to pass the same [`Decoder`]
7//! or [`Encoder`], also the ones which are otherwise ignored. The server only
8//! sends empty header blocks, the [`Encoder`] stores them uncompressed in the
9//! zlib stream instead of compressing them.
10
11use anyhow::{Context, Result, bail, ensure};
12use bytes::{Buf, BufMut, Bytes, BytesMut};
13use flate2::{Decompress, FlushDecompress, Status};
14use tokio::io::{AsyncRead, AsyncReadExt};
15
16/// The protocol version of the control frames.
17const VERSION: u16 = 3;
18
19/// The length of the common header of all frames.
20const HEADER_LEN: usize = 8;
21
22const TYPE_SYN_STREAM: u16 = 1;
23const TYPE_SYN_REPLY: u16 = 2;
24const TYPE_RST_STREAM: u16 = 3;
25const TYPE_PING: u16 = 6;
26const TYPE_GOAWAY: u16 = 7;
27const TYPE_HEADERS: u16 = 8;
28
29/// The flag which ends the sending side of a stream, on SYN_STREAM, SYN_REPLY,
30/// HEADERS and data frames.
31const FLAG_FIN: u8 = 0x01;
32
33/// The largest control frame a client may send. The ones Kubernetes clients
34/// send carry a few short headers.
35const MAX_CONTROL_FRAME_LEN: usize = 64 * 1024;
36
37/// The largest frame the protocol allows, its length has 24 bits.
38pub const MAX_DATA_FRAME_LEN: usize = 0xff_ffff;
39
40/// How much of a data frame the server reads at once. Larger frames get
41/// handed to their stream in parts, so that the server does not buffer them
42/// as a whole. Go clients send 32 KiB per frame.
43const DATA_CHUNK_LEN: usize = 64 * 1024;
44
45/// The largest decompressed header block, which bounds what a small
46/// compressed block may expand to.
47const MAX_HEADER_BLOCK_LEN: usize = 64 * 1024;
48
49/// RST_STREAM status: the frame violates the protocol.
50pub const STATUS_PROTOCOL_ERROR: u32 = 1;
51
52/// RST_STREAM status: the stream got refused before any processing.
53pub const STATUS_REFUSED_STREAM: u32 = 3;
54
55/// RST_STREAM status: the stream is not needed anymore. spdystream resets
56/// with it as well, also for streams it rejects.
57pub const STATUS_CANCEL: u32 = 5;
58
59/// GOAWAY status: the connection ends normally.
60pub const GOAWAY_OK: u32 = 0;
61
62/// GOAWAY status: the client violated the protocol.
63pub const GOAWAY_PROTOCOL_ERROR: u32 = 1;
64
65/// A single SPDY frame. Only the frames and fields the server acts on are
66/// represented, see [`Frame::Ignored`].
67#[derive(Debug, PartialEq)]
68pub enum Frame {
69    /// The client opens a stream.
70    SynStream {
71        stream_id: u32,
72        fin: bool,
73        headers: Headers,
74    },
75
76    /// The server accepts a stream.
77    SynReply { stream_id: u32, headers: Headers },
78
79    /// Either side aborts a stream.
80    RstStream { stream_id: u32, status: u32 },
81
82    /// A ping, which the receiver echoes if the other side started it.
83    Ping { id: u32 },
84
85    /// Either side ends the connection.
86    GoAway { last_stream_id: u32, status: u32 },
87
88    /// Additional headers, which only matter for their FIN flag.
89    Headers { stream_id: u32, fin: bool },
90
91    /// Data of a stream, an empty one with FIN closes its sending side.
92    Data {
93        stream_id: u32,
94        fin: bool,
95        data: Bytes,
96    },
97
98    /// A frame which needs no handling: SETTINGS, WINDOW_UPDATE or an unknown
99    /// control frame. spdystream does not do flow control, so window updates
100    /// are not needed either.
101    Ignored,
102}
103
104/// The headers of a stream. SPDY requires lowercase names, multiple values
105/// of a name are joined with NUL bytes.
106#[derive(Debug, Default, PartialEq)]
107pub struct Headers(Vec<(String, String)>);
108
109impl Headers {
110    /// The first value of a header, the name is matched case insensitively.
111    pub fn get(&self, name: &str) -> Option<&str> {
112        self.0
113            .iter()
114            .find(|(n, _)| n.eq_ignore_ascii_case(name))
115            .and_then(|(_, v)| v.split('\0').next())
116    }
117}
118
119impl<const N: usize> From<[(&str, &str); N]> for Headers {
120    fn from(headers: [(&str, &str); N]) -> Self {
121        Self(
122            headers
123                .into_iter()
124                .map(|(n, v)| (n.to_ascii_lowercase(), v.to_owned()))
125                .collect(),
126        )
127    }
128}

130/// Reads the frames of one direction of a connection.
131pub struct Decoder {
132    decompress: Decompress,
133
134    /// The data frame which is handed out in parts: its stream, the bytes
135    /// still to read and its FIN flag.
136    partial_data: Option<(u32, usize, bool)>,
137}
138
139impl Default for Decoder {
140    fn default() -> Self {
141        Self {
142            decompress: Decompress::new(true),
143            partial_data: None,
144        }
145    }
146}
147
148impl Decoder {
149    /// Read the next frame, `None` if the connection ended in between frames.
150    /// A data frame larger than [`DATA_CHUNK_LEN`] comes as several ones of
151    /// the same stream, only the last one with its FIN flag.
152    pub async fn read<R>(&mut self, reader: &mut R) -> Result<Option<Frame>>
153    where
154        R: AsyncRead + Unpin,
155    {
156        if let Some((stream_id, remaining, fin)) = self.partial_data {
157            return self
158                .read_data(reader, stream_id, remaining, fin)
159                .await
160                .map(Some);
161        }
162
163        let mut header = [0; HEADER_LEN];
164        let mut filled = 0;
165        while filled < HEADER_LEN {
166            match reader
167                .read(&mut header[filled..])
168                .await
169                .context("read frame header")?
170            {
171                0 if filled == 0 => return Ok(None),
172                0 => bail!("connection ended within a frame header"),
173                n => filled += n,
174            }
175        }
176
177        let control = header[0] & 0x80 != 0;
178        let flags = header[4];
179        let len =
180            usize::from(header[5]) << 16 | usize::from(header[6]) << 8 | usize::from(header[7]);
181        if !control && len > DATA_CHUNK_LEN {
182            let stream_id = u32::from_be_bytes([header[0], header[1], header[2], header[3]]);
183            return self
184                .read_data(reader, stream_id, len, flags & FLAG_FIN != 0)
185                .await
186                .map(Some);
187        }
188        ensure!(
189            len <= MAX_CONTROL_FRAME_LEN,
190            "frame of {len} bytes exceeds {MAX_CONTROL_FRAME_LEN} bytes"
191        );
192
193        let mut payload = BytesMut::zeroed(len);
194        reader
195            .read_exact(&mut payload)
196            .await
197            .context("read frame payload")?;
198        self.decode(header, flags, payload.freeze()).map(Some)
199    }
200
201    /// Read the next part of a large data frame.
202    async fn read_data<R>(
203        &mut self,
204        reader: &mut R,
205        stream_id: u32,
206        remaining: usize,
207        fin: bool,
208    ) -> Result<Frame>
209    where
210        R: AsyncRead + Unpin,
211    {
212        let len = remaining.min(DATA_CHUNK_LEN);
213        let mut data = BytesMut::zeroed(len);
214        reader
215            .read_exact(&mut data)
216            .await
217            .context("read frame payload")?;
218        let remaining = remaining - len;
219        self.partial_data = (remaining > 0).then_some((stream_id, remaining, fin));
220        Ok(Frame::Data {
221            stream_id,
222            fin: fin && remaining == 0,
223            data: data.freeze(),
224        })
225    }
226
227    /// Decode a frame from its header and complete payload.
228    fn decode(&mut self, header: [u8; HEADER_LEN], flags: u8, mut payload: Bytes) -> Result<Frame> {
229        let fin = flags & FLAG_FIN != 0;
230        let first = u32::from_be_bytes([header[0], header[1], header[2], header[3]]);
231
232        if first & 0x8000_0000 == 0 {
233            return Ok(Frame::Data {
234                stream_id: first,
235                fin,
236                data: payload,
237            });
238        }
239
240        let version = (first >> 16) as u16 & 0x7fff;
241        ensure!(version == VERSION, "unsupported SPDY version {version}");
242
243        let frame = match first as u16 {
244            TYPE_SYN_STREAM => {
245                // The stream ID, the associated stream ID, the priority and
246                // the slot precede the header block.
247                ensure!(payload.len() >= 10, "SYN_STREAM too short");
248                let stream_id = payload.get_u32() & 0x7fff_ffff;
249                payload.advance(6);
250                let headers = self.header_block(&payload)?;
251                Frame::SynStream {
252                    stream_id,
253                    fin,
254                    headers,
255                }
256            }
257            TYPE_SYN_REPLY => {
258                ensure!(payload.len() >= 4, "SYN_REPLY too short");
259                let stream_id = payload.get_u32() & 0x7fff_ffff;
260                let headers = self.header_block(&payload)?;
261                Frame::SynReply { stream_id, headers }
262            }
263            TYPE_RST_STREAM => {
264                ensure!(payload.len() == 8, "RST_STREAM has the wrong length");
265                Frame::RstStream {
266                    stream_id: payload.get_u32() & 0x7fff_ffff,
267                    status: payload.get_u32(),
268                }
269            }
270            TYPE_PING => {
271                ensure!(payload.len() == 4, "PING has the wrong length");
272                Frame::Ping {
273                    id: payload.get_u32(),
274                }
275            }
276            TYPE_GOAWAY => {
277                ensure!(payload.len() == 8, "GOAWAY has the wrong length");
278                Frame::GoAway {
279                    last_stream_id: payload.get_u32() & 0x7fff_ffff,
280                    status: payload.get_u32(),
281                }
282            }
283            TYPE_HEADERS => {
284                ensure!(payload.len() >= 4, "HEADERS too short");
285                let stream_id = payload.get_u32() & 0x7fff_ffff;
286                self.header_block(&payload)?;
287                Frame::Headers { stream_id, fin }
288            }
289            _ => Frame::Ignored,
290        };
291        Ok(frame)
292    }
293
294    /// Decompress and parse a header block.
295    fn header_block(&mut self, mut input: &[u8]) -> Result<Headers> {
296        // One byte more than the limit, so that a block of exactly the limit
297        // still leaves room, which tells that it is complete.
298        let max_capacity = MAX_HEADER_BLOCK_LEN + 1;
299        let mut block = Vec::with_capacity((input.len() * 4).min(max_capacity));
300        loop {
301            if block.len() == block.capacity() {
302                ensure!(
303                    block.len() <= MAX_HEADER_BLOCK_LEN,
304                    "header block exceeds {MAX_HEADER_BLOCK_LEN} bytes"
305                );
306                // Exactly, a larger capacity would let the block grow past
307                // the limit.
308                block.reserve_exact((max_capacity - block.len()).min(4096));
309            }
310
311            let (total_in, total_out) = (self.decompress.total_in(), self.decompress.total_out());
312            let status =
313                match self
314                    .decompress
315                    .decompress_vec(input, &mut block, FlushDecompress::Sync)
316                {
317                    Ok(status) => status,
318                    Err(e) if e.needs_dictionary().is_some() => {
319                        self.decompress
320                            .set_dictionary(DICTIONARY)
321                            .context("set header dictionary")?;
322                        Status::Ok
323                    }
324                    Err(e) => return Err(e).context("decompress header block"),
325                };
326            input = &input[(self.decompress.total_in() - total_in) as usize..];
327            // The allocator may still have given more room than asked for.
328            ensure!(
329                block.len() <= MAX_HEADER_BLOCK_LEN,
330                "header block exceeds {MAX_HEADER_BLOCK_LEN} bytes"
331            );
332
333            // The sender flushes after each block, so all of it is there once
334            // the input is consumed and the output has room left.
335            if status == Status::StreamEnd || input.is_empty() && block.len() < block.capacity() {
336                break;
337            }
338            ensure!(
339                block.len() == block.capacity()
340                    || self.decompress.total_in() > total_in
341                    || self.decompress.total_out() > total_out,
342                "header block does not decompress"
343            );
344        }
345
346        Self::parse_header_block(&block)
347    }
348
349    fn parse_header_block(mut block: &[u8]) -> Result<Headers> {
350        fn string(block: &mut &[u8]) -> Result<String> {
351            ensure!(block.len() >= 4, "header block truncated");
352            let len = block.get_u32() as usize;
353            ensure!(block.len() >= len, "header block truncated");
354            let s = String::from_utf8_lossy(&block[..len]).into_owned();
355            block.advance(len);
356            Ok(s)
357        }
358
359        ensure!(block.len() >= 4, "header block truncated");
360        let count = block.get_u32() as usize;
361        // Each pair takes at least the two lengths.
362        ensure!(count <= block.len() / 8, "header block truncated");
363        let mut headers = Vec::with_capacity(count);
364        for _ in 0..count {
365            let name = string(&mut block)?;
366            let value = string(&mut block)?;
367            headers.push((name, value));
368        }
369        Ok(Headers(headers))
370    }
371}

373/// Writes the frames of one direction of a connection.
374#[derive(Default)]
375pub struct Encoder {
376    /// Whether the zlib header went out with the first header block.
377    started: bool,
378}
379
380impl Encoder {
381    /// Append the encoded frame to the buffer.
382    pub fn encode(&mut self, frame: &Frame, buf: &mut BytesMut) -> Result<()> {
383        match frame {
384            Frame::SynStream {
385                stream_id,
386                fin,
387                headers,
388            } => {
389                let block = self.header_block(headers);
390                Self::control(buf, TYPE_SYN_STREAM, *fin, 10 + block.len());
391                buf.put_u32(*stream_id);
392                // No associated stream, the lowest priority and no slot.
393                buf.put_u32(0);
394                buf.put_u16(0);
395                buf.put_slice(&block);
396            }
397            Frame::SynReply { stream_id, headers } => {
398                let block = self.header_block(headers);
399                Self::control(buf, TYPE_SYN_REPLY, false, 4 + block.len());
400                buf.put_u32(*stream_id);
401                buf.put_slice(&block);
402            }
403            Frame::RstStream { stream_id, status } => {
404                Self::control(buf, TYPE_RST_STREAM, false, 8);
405                buf.put_u32(*stream_id);
406                buf.put_u32(*status);
407            }
408            Frame::Ping { id } => {
409                Self::control(buf, TYPE_PING, false, 4);
410                buf.put_u32(*id);
411            }
412            Frame::GoAway {
413                last_stream_id,
414                status,
415            } => {
416                Self::control(buf, TYPE_GOAWAY, false, 8);
417                buf.put_u32(*last_stream_id);
418                buf.put_u32(*status);
419            }
420            Frame::Headers { stream_id, fin } => {
421                let block = self.header_block(&Headers::default());
422                Self::control(buf, TYPE_HEADERS, *fin, 4 + block.len());
423                buf.put_u32(*stream_id);
424                buf.put_slice(&block);
425            }
426            Frame::Data {
427                stream_id,
428                fin,
429                data,
430            } => {
431                ensure!(data.len() <= MAX_DATA_FRAME_LEN, "data frame too large");
432                buf.put_u32(*stream_id);
433                buf.put_u8(if *fin { FLAG_FIN } else { 0 });
434                Self::put_len(buf, data.len());
435                buf.put_slice(data);
436            }
437            Frame::Ignored => {}
438        }
439        Ok(())
440    }
441
442    fn control(buf: &mut BytesMut, typ: u16, fin: bool, len: usize) {
443        buf.put_u16(0x8000 | VERSION);
444        buf.put_u16(typ);
445        buf.put_u8(if fin { FLAG_FIN } else { 0 });
446        Self::put_len(buf, len);
447    }
448
449    fn put_len(buf: &mut BytesMut, len: usize) {
450        buf.put_slice(&(len as u32).to_be_bytes()[1..]);
451    }
452
453    /// Serialize a header block into stored deflate blocks, which every
454    /// zlib decompressor reads like compressed ones. Compressing the blocks
455    /// of the server would gain nothing, they are empty.
456    fn header_block(&mut self, headers: &Headers) -> Vec<u8> {
457        let mut raw = Vec::new();
458        raw.put_u32(headers.0.len() as u32);
459        for (name, value) in &headers.0 {
460            raw.put_u32(name.len() as u32);
461            raw.put_slice(name.as_bytes());
462            raw.put_u32(value.len() as u32);
463            raw.put_slice(value.as_bytes());
464        }
465        self.deflate(&raw)
466    }
467
468    /// Wrap a serialized header block into stored deflate blocks, the first
469    /// one of the connection with the zlib header.
470    fn deflate(&mut self, raw: &[u8]) -> Vec<u8> {
471        let mut block = Vec::with_capacity(raw.len() + 16);
472        if !self.started {
473            // Deflate with a 32 KiB window and the preset dictionary.
474            block.put_slice(&[0x78, 0x20]);
475            block.put_u32(DICTIONARY_ID);
476            self.started = true;
477        }
478        for chunk in raw.chunks(usize::from(u16::MAX)) {
479            Self::stored_block(&mut block, chunk);
480        }
481        // The empty block of a sync flush, Go hands out the data before it
482        // only once it read one. Without it, the decompressor reads beyond
483        // the frame and fails all later header blocks.
484        Self::stored_block(&mut block, &[]);
485        block
486    }
487
488    /// A stored deflate block which is not the last one.
489    fn stored_block(block: &mut Vec<u8>, data: &[u8]) {
490        let len = data.len() as u16;
491        block.put_u8(0);
492        block.put_u16_le(len);
493        block.put_u16_le(!len);
494        block.put_slice(data);
495    }
496}

504    /// Decode frames until the input ends or is invalid. Reading a slice is
505    /// never pending, so no runtime is needed.
506    pub fn frames(mut input: &[u8]) {
507        let mut decoder = Decoder::default();
508        while let Some(Ok(Some(_))) = decoder.read(&mut input).now_or_never() {}
509    }
510
511    /// Decode the input as a compressed header block, which is what a
512    /// client controls, parse it as a decompressed one, and decode it once
513    /// deflated like the server does, which always gets past the
514    /// decompression to the parsing.
515    pub fn header_block(raw: &[u8]) {
516        let _ = Decoder::default().header_block(raw);
517        let _ = Decoder::parse_header_block(raw);
518        let block = Encoder::default().deflate(raw);
519        let _ = Decoder::default().header_block(&block);
520    }

774    #[tokio::test]
775    async fn large_data_frames_come_in_parts() -> Result<()> {
776        // The largest frame the protocol allows, with FIN, then a ping.
777        let len = MAX_DATA_FRAME_LEN;
778        let mut bytes = vec![0, 0, 0, 1, FLAG_FIN];
779        bytes.extend_from_slice(&(len as u32).to_be_bytes()[1..]);
780        bytes.extend((0..len).map(|i| i as u8));
781        bytes.extend(hex(GO_PING));
782
783        let frames = decode_all(&mut Decoder::default(), &bytes).await?;
784        assert_eq!(frames.len(), len.div_ceil(DATA_CHUNK_LEN) + 1);
785        let mut received = vec![];
786        for (i, frame) in frames[..frames.len() - 1].iter().enumerate() {
787            let Frame::Data {
788                stream_id: 1,
789                fin,
790                data,
791            } = frame
792            else {
793                panic!("not a data frame of stream 1: {frame:?}");
794            };
795            assert!(data.len() <= DATA_CHUNK_LEN);
796            // Only the last part ends the stream.
797            assert_eq!(*fin, i == frames.len() - 2);
798            received.extend_from_slice(data);
799        }
800        assert!(received.iter().enumerate().all(|(i, b)| *b == i as u8));
801        assert_eq!(received.len(), len);
802        assert_eq!(frames.last(), Some(&Frame::Ping { id: 1 }));
803
804        // A frame which ends within its payload is an error.
805        let mut decoder = Decoder::default();
806        let truncated = &bytes[..HEADER_LEN + DATA_CHUNK_LEN + 1];
807        let mut reader = truncated;
808        decoder.read(&mut reader).await?;
809        assert!(decoder.read(&mut reader).await.is_err());
810        Ok(())
811    }

813    /// A HEADERS frame whose block decompresses to `len` zeros, which is a
814    /// block without headers followed by padding.
815    fn headers_frame(len: usize) -> Result<BytesMut> {
816        let mut compress = Compress::new(Compression::best(), true);
817        compress.set_dictionary(DICTIONARY)?;
818        let mut block = Vec::with_capacity(16 * 1024);
819        compress.compress_vec(&vec![0; len], &mut block, FlushCompress::Sync)?;
820        assert!(block.len() < MAX_CONTROL_FRAME_LEN);
821
822        let mut frame = BytesMut::new();
823        Encoder::control(&mut frame, TYPE_HEADERS, false, 4 + block.len());
824        frame.put_u32(1);
825        frame.put_slice(&block);
826        Ok(frame)
827    }
